### 3.3 启动脚本

创建 `/opt/musheng/start.sh`:

```bash
#!/bin/bash

APP_NAME="musheng-web"
JAR_FILE="/opt/musheng/musheng-web-1.0.0-SNAPSHOT.jar"
LOG_DIR="/var/log/musheng"
PID_FILE="/var/run/musheng.pid"

# 环境变量
export JAVA_HOME=/usr/lib/jvm/java-17-openjdk
export DB_USERNAME=musheng
export DB_PASSWORD=YourSecurePassword
export REDIS_HOST=localhost
export REDIS_PASSWORD=YourRedisPassword

# JVM参数
JAVA_OPTS="-Xms512m -Xmx2g -XX:+UseG1GC -XX:MaxGCPauseMillis=200"
JAVA_OPTS="$JAVA_OPTS -Dspring.profiles.active=prod"
JAVA_OPTS="$JAVA_OPTS -Dfile.encoding=UTF-8"

# 创建日志目录
mkdir -p $LOG_DIR

# 启动函数
start() {
    if [ -f $PID_FILE ]; then
        PID=$(cat $PID_FILE)
        if ps -p $PID > /dev/null; then
            echo "$APP_NAME is already running (PID: $PID)"
            return 1
        fi
    fi

    echo "Starting $APP_NAME..."
    nohup $JAVA_HOME/bin/java $JAVA_OPTS -jar $JAR_FILE > $LOG_DIR/console.log 2>&1 &
    echo $! > $PID_FILE
    echo "$APP_NAME started (PID: $(cat $PID_FILE))"
}

# 停止函数
stop() {
    if [ -f $PID_FILE ]; then
        PID=$(cat $PID_FILE)
        if ps -p $PID > /dev/null; then
            echo "Stopping $APP_NAME (PID: $PID)..."
            kill $PID
            sleep 5
            if ps -p $PID > /dev/null; then
                echo "Force killing..."
                kill -9 $PID
            fi
            rm -f $PID_FILE
            echo "$APP_NAME stopped"
        else
            echo "$APP_NAME is not running"
            rm -f $PID_FILE
        fi
    else
        echo "$APP_NAME is not running"
    fi
}

# 重启函数
restart() {
    stop
    sleep 2
    start
}

# 状态函数
status() {
    if [ -f $PID_FILE ]; then
        PID=$(cat $PID_FILE)
        if ps -p $PID > /dev/null; then
            echo "$APP_NAME is running (PID: $PID)"
        else
            echo "$APP_NAME is not running (stale PID file)"
        fi
    else
        echo "$APP_NAME is not running"
    fi
}

case "$1" in
    start)
        start
        ;;
    stop)
        stop
        ;;
    restart)
        restart
        ;;
    status)
        status
        ;;
    *)
        echo "Usage: $0 {start|stop|restart|status}"
        exit 1
        ;;
esac
```

```bash
# 赋予执行权限
chmod +x /opt/musheng/start.sh

# 启动服务
/opt/musheng/start.sh start

# 查看状态
/opt/musheng/start.sh status
```

### 3.4 Systemd服务配置

创建 `/etc/systemd/system/musheng.service`:

```ini
[Unit]
Description=Musheng Tax System
After=network.target mysql.service redis.service

[Service]
Type=forking
User=musheng
Group=musheng
Environment="JAVA_HOME=/usr/lib/jvm/java-17-openjdk"

ExecReload=/opt/musheng/start.sh restart
PIDFile=/var/run/musheng.pid
Restart=on-failure
RestartSec=10

[Install]
WantedBy=multi-user.target
```

### 4.2 Nginx配置

创建 `/etc/nginx/conf.d/musheng.conf`:

```nginx
# HTTP重定向到HTTPS
server {
    listen 80;
    server_name tax.musheng.com;
    return 301 https://$server_name$request_uri;
}

# HTTPS服务
server {
    listen 443 ssl http2;
    server_name tax.musheng.com;

    # SSL证书配置
    ssl_certificate /etc/nginx/ssl/musheng.com.pem;
    ssl_certificate_key /etc/nginx/ssl/musheng.com.key;
    ssl_session_timeout 5m;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:HIGH:!aNULL:!MD5:!RC4:!DHE;
    ssl_prefer_server_ciphers on;

    # 前端静态资源
    root /var/www/musheng/dist;
    index index.html;

    # Gzip压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/javascript application/javascript application/json application/xml;

    # 静态资源缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # 前端路由 (Vue Router History模式)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # API代理
    location /api/ {
        proxy_pass http://127.0.0.1:8080/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 文件上传大小限制
        client_max_body_size 100m;

        # 超时设置
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    # API文档
    location /doc.html {
        proxy_pass http://127.0.0.1:8080/doc.html;
        proxy_set_header Host $host;
    }

    # WebSocket支持（如需要）
    location /ws/ {
        proxy_pass http://127.0.0.1:8080/ws/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    # 错误页面
    error_page 500 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;
    }

    # 访问日志
    access_log /var/log/nginx/musheng_access.log;
    error_log /var/log/nginx/musheng_error.log;
}
```

```bash

# 重载Nginx

```

### 4.3 部署脚本

创建 `/opt/musheng/deploy-web.sh`:

```bash
#!/bin/bash

WEB_ROOT="/var/www/musheng"
BACKUP_DIR="/var/www/musheng-backup"
DIST_FILE="$1"

if [ -z "$DIST_FILE" ]; then
    echo "Usage: $0 <dist.tar.gz>"
    exit 1
fi

# 备份当前版本
if [ -d "$WEB_ROOT/dist" ]; then
    TIMESTAMP=$(date +%Y%m%d_%H%M%S)
    mkdir -p $BACKUP_DIR
    mv $WEB_ROOT/dist $BACKUP_DIR/dist_$TIMESTAMP
    echo "Backed up to $BACKUP_DIR/dist_$TIMESTAMP"
fi

# 解压新版本
mkdir -p $WEB_ROOT
tar -xzf $DIST_FILE -C $WEB_ROOT/

# 设置权限
chown -R nginx:nginx $WEB_ROOT
chmod -R 755 $WEB_ROOT

# 重载Nginx
nginx -s reload

echo "Deployment completed!"
```

## 七、常见问题

### 7.1 后端启动失败

**问题**: 启动时报数据库连接失败

**解决方案**:
```bash
# 检查MySQL状态
systemctl status mysql

# 检查连接
mysql -u musheng -p -h localhost musheng_tax

# 检查配置文件中的连接字符串
cat application-prod.yml | grep url
```

### 7.2 前端页面空白

**问题**: 访问页面显示空白

**解决方案**:
```bash
# 检查Nginx配置
nginx -t

# 检查前端文件
ls -la /var/www/musheng/dist/

# 检查浏览器控制台错误
# 通常是API地址配置错误
```

### 7.3 API请求失败

**问题**: 前端调用API返回502

**解决方案**:
```bash
# 检查后端服务状态
curl http://localhost:8080/actuator/health

# 检查Nginx代理配置
cat /etc/nginx/conf.d/musheng.conf | grep proxy_pass

# 检查后端日志
tail -f /var/log/musheng/app.log
```

### 7.4 文件上传失败

**问题**: 大文件上传超时

**解决方案**:
```nginx
# Nginx配置增加
client_max_body_size 100m;
proxy_connect_timeout 300s;
proxy_send_timeout 300s;
proxy_read_timeout 300s;
```

### 7.5 内存不足

**问题**: 服务频繁OOM

**解决方案**:
```bash
# 调整JVM参数
JAVA_OPTS="-Xms1g -Xmx2g -XX:+UseG1GC"

# 检查内存使用
free -h
top -p $(cat /var/run/musheng.pid)
```
