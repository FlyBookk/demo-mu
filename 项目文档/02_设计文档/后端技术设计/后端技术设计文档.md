## 2. 模块设计

### 2.1 模块职责说明

| 模块名称 | 职责说明 | 核心功能 |
|---------|---------|---------|
| musheng-common | 公共基础模块 | 异常处理、统一响应、工具类、常量枚举 |
| musheng-system | 系统管理模块 | 用户管理、角色权限、操作日志、登录认证 |
| musheng-config | 基础配置模块 | 货币管理、站点管理、字段映射、交易类型映射、导入记录 |
| musheng-business | 业务数据模块 | 销售数据、配送数据、广告数据、汇率数据、汇总报表 |
| musheng-web | Web启动模块 | 应用启动、全局配置、安全配置 |

### 2.2 模块依赖关系

```
musheng-web
    ├── musheng-system
    ├── musheng-config
    └── musheng-business
            └── musheng-config
                    └── musheng-common

依赖方向: 上层依赖下层，禁止循环依赖
```

```sql
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
```

### 4.2 统一响应格式

```java
/**
 * 统一响应结果
 */
@Data
public class Result<T> {
    /** 状态码 */
    private int code;
    /** 消息 */
    private String message;
    /** 数据 */
    private T data;
    /** 时间戳 */
    private long timestamp;
    /** 请求ID(用于链路追踪) */
    private String requestId;

    public static <T> Result<T> success(T data) {
        Result<T> result = new Result<>();
        result.setCode(200);
        result.setMessage("success");
        result.setData(data);
        result.setTimestamp(System.currentTimeMillis());
        return result;
    }

    public static <T> Result<T> error(int code, String message) {
        Result<T> result = new Result<>();
        result.setCode(code);
        result.setMessage(message);
        result.setTimestamp(System.currentTimeMillis());
        return result;
    }
}

/**
 * 分页响应结果
 */
@Data
public class PageResult<T> {
    private List<T> records;
    private long total;
    private int page;
    private int size;
    private int pages;
}
```

```java
@Service
public class HeaderDetectorService {

    // 关键表头字段映射 - 用于识别表头行
    private static final Map<String, String> HEADER_KEYWORDS = Map.of(
        "EN", "date/time,settlement id,type,order id",
        "DE", "Datum/Uhrzeit,Abrechnungsnummer,Typ,Bestellnummer"
    );

    /**
     * 识别表头行
     * @param lines CSV文件行列表
     * @return 表头行号(0-based)，未找到返回-1
     */
    public int detectHeaderRow(List<String[]> lines) {
        for (int i = 0; i < Math.min(lines.size(), 20); i++) {
            String lineStr = String.join(",", lines.get(i)).toLowerCase();
            for (String keywords : HEADER_KEYWORDS.values()) {
                String[] keys = keywords.split(",");
                boolean allMatch = Arrays.stream(keys)
                    .allMatch(k -> lineStr.contains(k.toLowerCase()));
                if (allMatch) {
                    return i;
                }
            }
        }
        return -1;
    }

    /**
     * 检测表头语言
     */
    public String detectLanguage(String[] headers) {
        String headerStr = String.join(",", headers).toLowerCase();
        if (headerStr.contains("datum/uhrzeit") || headerStr.contains("bestellnummer")) {
            return "DE";
        }
        return "EN";
    }
```

```java
@Service
@Slf4j
public class DateParseService {

    /**
     * 解析日期字符串
     * @param dateStr 原始日期字符串
     * @param siteCode 站点编码
     * @return 解析后的LocalDateTime
     */
    public LocalDateTime parseDate(String dateStr, String siteCode) {
        if (StringUtils.isBlank(dateStr)) {
            return null;
        }

        // 预处理 - 加拿大站特殊处理 (a.m./p.m. 带点格式)
        String processedStr = preprocessDateString(dateStr, siteCode);

        // 根据站点获取日期格式
        String pattern = getDatePattern(siteCode);
        Locale locale = getLocale(siteCode);

        try {
            DateTimeFormatter formatter = DateTimeFormatter.ofPattern(pattern, locale);
            return LocalDateTime.parse(processedStr, formatter);
        } catch (DateTimeParseException e) {
            log.warn("日期解析失败: dateStr={}, siteCode={}, pattern={}",
                dateStr, siteCode, pattern);
            // 尝试备用格式
            return tryAlternativePatterns(processedStr, siteCode);
        }
    }

    /**
     * 预处理日期字符串
     * 主要处理加拿大站的 a.m./p.m. 格式
     */
    private String preprocessDateString(String dateStr, String siteCode) {
        if ("CA".equals(siteCode)) {
            // 加拿大站: "Jul 1, 2025 12:58:35 a.m. PDT" -> "Jul 1, 2025 12:58:35 AM PDT"
            return dateStr
                .replace("a.m.", "AM")
                .replace("p.m.", "PM")
                .replace("A.M.", "AM")
                .replace("P.M.", "PM");
        }
        return dateStr;
    }

    /**
     * 获取站点对应的日期格式
     */
    private String getDatePattern(String siteCode) {
        return switch (siteCode) {
            case "US", "CA" -> "MMM d, yyyy h:mm:ss a z";
            case "UK" -> "dd MMM yyyy HH:mm:ss z";
            case "DE" -> "dd.MM.yyyy HH:mm:ss z";
            default -> "MMM d, yyyy h:mm:ss a z";
        };
    }

    /**
     * 获取站点对应的Locale
     */
    private Locale getLocale(String siteCode) {
        return switch (siteCode) {
            case "DE" -> Locale.GERMAN;
            default -> Locale.ENGLISH;
        };
    }

    /**
     * 尝试备用日期格式
     */
    private LocalDateTime tryAlternativePatterns(String dateStr, String siteCode) {
        List<String> patterns = List.of(
            "yyyy-MM-dd HH:mm:ss",
            "yyyy/MM/dd HH:mm:ss",
            "MM/dd/yyyy HH:mm:ss",
            "dd/MM/yyyy HH:mm:ss"
        );

        for (String pattern : patterns) {
            try {
                DateTimeFormatter formatter = DateTimeFormatter.ofPattern(pattern);
                return LocalDateTime.parse(dateStr.substring(0, 19), formatter);
            } catch (Exception ignored) {
            }
        }

        throw new BusinessException(ErrorCode.IMPORT_FILE_FORMAT_ERROR,
            "无法解析日期: " + dateStr);
    }
}
```

```java
            return 0;
        }
    }
```

```java
    /**
     * 根据映射配置转换数据
     * @param sourceRow 源数据行
     * @param sourceHeaders 源表头
     * @param mappingConfig 映射配置
     * @return 转换后的Map
     */
    public Map<String, Object> mapFields(String[] sourceRow, String[] sourceHeaders,
            MappingConfig mappingConfig) {

        Map<String, Object> result = new HashMap<>();
        Map<String, Integer> headerIndex = buildHeaderIndex(sourceHeaders);

        // 应用字段映射
        for (FieldMapping mapping : mappingConfig.getMappings()) {
            String sourceField = mapping.getSource();
            String targetField = mapping.getTarget();

            Integer idx = headerIndex.get(sourceField.toLowerCase());
            if (idx != null && idx < sourceRow.length) {
                result.put(targetField, sourceRow[idx]);
            }
        }

        // 应用默认值
        if (mappingConfig.getUnmappedDefaults() != null) {
            mappingConfig.getUnmappedDefaults().forEach((key, value) -> {
                if (!result.containsKey(key)) {
                    result.put(key, value);
                }
            });
        }

        return result;
    }

    /**
     * 获取站点默认映射模板
     */
    public FieldMappingTemplate getDefaultTemplate(String siteCode, String dataType) {
        return templateMapper.selectOne(

                .eq(FieldMappingTemplate::getDataType, dataType)
                .eq(FieldMappingTemplate::getIsDefault, 1)
```

### 5.3 汇率服务(节假日顺延逻辑)

```java
/**
 * 汇率服务
 * 实现节假日顺延逻辑
 */
@Service
@Slf4j
public class ExchangeRateServiceImpl implements ExchangeRateService {

    @Autowired
    private ExchangeRateMapper rateMapper;

    @Autowired
    private HolidayMapper holidayMapper;

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    // 汇率顺延最大天数
    private static final int MAX_DEFER_DAYS = 10;

    /**
     * 获取指定日期的汇率
     * 节假日自动顺延到下一个工作日
     *
     * @param date 日期
     * @param currencyCode 货币编码
     * @return 汇率
     */
    @Override
    public BigDecimal getRate(LocalDate date, String currencyCode) {
        // 1. 先查缓存
        String cacheKey = String.format("rate:%s:%s", currencyCode, date);
        BigDecimal cachedRate = (BigDecimal) redisTemplate.opsForValue().get(cacheKey);
        if (cachedRate != null) {
            return cachedRate;
        }

        // 2. 查询实际汇率日期(处理节假日顺延)
        LocalDate actualDate = findActualRateDate(date);

        // 3. 查询汇率
        ExchangeRate rate = rateMapper.selectOne(
            new LambdaQueryWrapper<ExchangeRate>()
                .eq(ExchangeRate::getRateDate, actualDate)
                .eq(ExchangeRate::getCurrencyCode, currencyCode)
        );

        if (rate == null) {
            throw new BusinessException(ErrorCode.RATE_NOT_FOUND,
                String.format("汇率不存在: %s - %s", actualDate, currencyCode));
        }

        // 4. 缓存结果
        redisTemplate.opsForValue().set(cacheKey, rate.getRate(), 24, TimeUnit.HOURS);

        return rate.getRate();
    }

    /**
     * 查找实际汇率日期(节假日顺延)
     */
    private LocalDate findActualRateDate(LocalDate date) {
        LocalDate currentDate = date;
        int deferDays = 0;

        while (deferDays < MAX_DEFER_DAYS) {
            // 检查是否为工作日
            if (isWorkday(currentDate)) {
                return currentDate;
            }
            // 顺延到下一天
            currentDate = currentDate.plusDays(1);
            deferDays++;
        }

        // 超过最大顺延天数，返回原日期
        log.warn("汇率顺延超过最大天数: date={}, maxDays={}", date, MAX_DEFER_DAYS);
        return date;
    }

    /**
     * 判断是否为工作日
     */
    private boolean isWorkday(LocalDate date) {
        // 1. 检查是否为周末
        DayOfWeek dayOfWeek = date.getDayOfWeek();
        if (dayOfWeek == DayOfWeek.SATURDAY || dayOfWeek == DayOfWeek.SUNDAY) {
            return false;
        }

        // 2. 检查是否为法定节假日
        Long holidayCount = holidayMapper.selectCount(
            new LambdaQueryWrapper<Holiday>()
                .eq(Holiday::getHolidayDate, date)
        );

        return holidayCount == 0;
    }

    /**
     * 批量导入汇率数据
     * 自动填充节假日空白日期
     */
    @Override
    @Transactional(rollbackFor = Exception.class)
    public ImportResult importRates(MultipartFile file) {
        List<ExchangeRate> rates = parseRateFile(file);

        // 按货币分组
        Map<String, List<ExchangeRate>> ratesByCode = rates.stream()
            .collect(Collectors.groupingBy(ExchangeRate::getCurrencyCode));

        int successCount = 0;
        int failCount = 0;

        for (Map.Entry<String, List<ExchangeRate>> entry : ratesByCode.entrySet()) {
            String currencyCode = entry.getKey();
            List<ExchangeRate> currencyRates = entry.getValue();

            // 按日期排序
            currencyRates.sort(Comparator.comparing(ExchangeRate::getRateDate));

            // 填充空白日期(节假日使用下一工作日汇率)
            List<ExchangeRate> filledRates = fillHolidayRates(currencyRates);

            // 批量保存
            for (ExchangeRate rate : filledRates) {
                try {
                    rateMapper.insertOrUpdate(rate);
                    successCount++;
                } catch (Exception e) {
                    log.error("汇率保存失败: {}", rate, e);
                    failCount++;
                }
            }
        }

        // 清除汇率缓存
        clearRateCache();

        return ImportResult.builder()
            .successCount(successCount)
            .failCount(failCount)
            .build();
    }

    /**
     * 填充节假日汇率
     * 节假日使用顺延后的下一工作日汇率
     */
    private List<ExchangeRate> fillHolidayRates(List<ExchangeRate> workdayRates) {
        if (workdayRates.isEmpty()) {
            return workdayRates;
        }

        List<ExchangeRate> result = new ArrayList<>();
        String currencyCode = workdayRates.get(0).getCurrencyCode();

        LocalDate startDate = workdayRates.get(0).getRateDate();
        LocalDate endDate = workdayRates.get(workdayRates.size() - 1).getRateDate();

        // 构建工作日汇率Map
        Map<LocalDate, BigDecimal> workdayRateMap = workdayRates.stream()
            .collect(Collectors.toMap(
                ExchangeRate::getRateDate,
                ExchangeRate::getRate
            ));

        // 遍历日期范围，填充节假日
        LocalDate currentDate = startDate;
        while (!currentDate.isAfter(endDate)) {
            ExchangeRate rate = new ExchangeRate();
            rate.setCurrencyCode(currencyCode);
            rate.setRateDate(currentDate);

            if (workdayRateMap.containsKey(currentDate)) {
                // 工作日，直接使用
                rate.setRate(workdayRateMap.get(currentDate));
                rate.setIsWorkday(1);
                rate.setActualRateDate(currentDate);
            } else {
                // 节假日，查找下一工作日汇率
                LocalDate nextWorkday = findNextWorkdayWithRate(currentDate, workdayRateMap);
                if (nextWorkday != null) {
                    rate.setRate(workdayRateMap.get(nextWorkday));
                    rate.setIsWorkday(0);
                    rate.setActualRateDate(nextWorkday);
                }
            }

            if (rate.getRate() != null) {
                result.add(rate);
            }

            currentDate = currentDate.plusDays(1);
        }

        return result;
    }

    /**
     * 查找下一个有汇率的工作日
     */
    private LocalDate findNextWorkdayWithRate(LocalDate date, Map<LocalDate, BigDecimal> rateMap) {
        LocalDate current = date.plusDays(1);
        int maxDays = MAX_DEFER_DAYS;

        while (maxDays > 0) {
            if (rateMap.containsKey(current)) {
                return current;
            }
            current = current.plusDays(1);
            maxDays--;
        }

        return null;
    }

    /**
     * 清除汇率缓存
     */
    private void clearRateCache() {
        Set<String> keys = redisTemplate.keys("rate:*");
        if (keys != null && !keys.isEmpty()) {
            redisTemplate.delete(keys);
        }
    }
}
```

```java
@Service

public class SummaryServiceImpl implements SummaryService {

    /**
     * 计算单个站点汇总
     */
    private SummaryResult calculateSiteSummary(Marketplace site, int year, String quarter,
            LocalDate startDate, LocalDate endDate) {

        String siteCode = site.getSiteCode();
        String currencyCode = site.getCurrencyCode();

        // 1. 计算收入总额(基于配送数据)

        BigDecimal netIncome = totalRevenue.subtract(refundAmount);

        // 4. 计算亚马逊服务费
        BigDecimal serviceFees = calculateServiceFees(siteCode, startDate, endDate, currencyCode);

        // 5. 计算广告费
        BigDecimal advertisingFees = calculateAdvertisingFees(siteCode, year, quarter);

        // 6. 计算平台费用总额
        BigDecimal platformFees = serviceFees.add(advertisingFees);

        // 7. 统计订单数量
        int orderCount = countOrders(siteCode, startDate, endDate);

        return SummaryResult.builder()
            .mcid(site.getSellerId())
            .marketplace(site.getSiteName())
            .companyName("dongguanshimushengshangmaoyouxiangongsi")
            .taxId("91441900MA4WNG4C6H")
            .quarter(quarter)
            .totalRevenue(totalRevenue.setScale(2, RoundingMode.HALF_UP))
            .refundAmount(refundAmount.setScale(2, RoundingMode.HALF_UP))
            .netIncome(netIncome.setScale(2, RoundingMode.HALF_UP))
            .platformFees(platformFees.setScale(2, RoundingMode.HALF_UP))
            .transactionCount(orderCount)
            .build();
    }

    /**
     * 计算收入总额(基于配送数据)
     * 收入总额 = SUM(商品价格 + 商品税 + 运费 + 运费税 + 礼品包装价格 + 礼品包装价格税
     *           + 商品促销折扣 + 货件促销折扣) × 配送日期汇率
     */
    private BigDecimal calculateTotalRevenue(String siteCode, LocalDate startDate,
            LocalDate endDate, String currencyCode) {

        List<ShippingData> shippingList = shippingDataMapper.selectList(
            new LambdaQueryWrapper<ShippingData>()
                .eq(ShippingData::getSiteCode, siteCode)
                .ge(ShippingData::getShipDate, startDate)
                .le(ShippingData::getShipDate, endDate)
        );

        BigDecimal total = BigDecimal.ZERO;

        for (ShippingData shipping : shippingList) {
            // 计算单条配送收入
            BigDecimal revenue = shipping.getProductPrice()
                .add(shipping.getProductTax())
                .add(shipping.getShippingPrice())
                .add(shipping.getShippingTax())
                .add(shipping.getGiftWrapPrice())
                .add(shipping.getGiftWrapTax())
                .add(shipping.getProductPromotionDiscount())
                .add(shipping.getShipmentPromotionDiscount());

            // 获取配送日期汇率
            BigDecimal rate = rateService.getRate(shipping.getShipDate(), currencyCode);

            // 换算为人民币
            total = total.add(revenue.multiply(rate));
        }

        return total;
    }

    /**
     * 计算退款金额
     * 根据订单号查找配送日期，以配送日期归属月份
     */
    private BigDecimal calculateRefundAmount(String siteCode, LocalDate startDate,
            LocalDate endDate, String currencyCode) {

        // 查询退款数据(时间范围需要延长2个月以捕获跨月退款)
        LocalDate extendedEndDate = endDate.plusMonths(2);

        List<SalesData> refunds = salesDataMapper.selectList(
            new LambdaQueryWrapper<SalesData>()
                .eq(SalesData::getSiteCode, siteCode)
                .eq(SalesData::getTransactionCategory, "refund")
                .ge(SalesData::getTransactionDate, startDate)
                .le(SalesData::getTransactionDate, extendedEndDate)
        );

        BigDecimal total = BigDecimal.ZERO;

        for (SalesData refund : refunds) {
            // 根据订单号查找配送日期
            ShippingData shipping = shippingDataMapper.selectOne(
                new LambdaQueryWrapper<ShippingData>()
                    .eq(ShippingData::getOrderId, refund.getOrderId())
                    .last("LIMIT 1")
            );

            if (shipping == null) {
                log.warn("退款订单找不到配送记录: orderId={}", refund.getOrderId());
                continue;
            }

            // 判断配送日期是否在目标季度范围内
            LocalDate shipDate = shipping.getShipDate();
            if (shipDate.isBefore(startDate) || shipDate.isAfter(endDate)) {
                continue;
            }

            // 计算退款金额
            BigDecimal refundValue = refund.getProductSales().abs()
                .add(refund.getProductSalesTax().abs())
                .add(refund.getShippingCredits().abs())
                .add(refund.getShippingCreditsTax().abs())
                .add(refund.getGiftWrapCredits().abs())
                .add(refund.getGiftWrapCreditsTax().abs())
                .add(refund.getPromotionalRebates().abs())
                .add(refund.getPromotionalRebatesTax().abs());

            // 使用配送日期汇率
            BigDecimal rate = rateService.getRate(shipDate, currencyCode);

            total = total.add(refundValue.multiply(rate));
        }

        return total;
    }

    /**
     * 计算亚马逊服务费
     * 服务费 = SUM(|销售费用| + |FBA费用| + |其他交易费|) × 当日汇率
     */
    private BigDecimal calculateServiceFees(String siteCode, LocalDate startDate,
            LocalDate endDate, String currencyCode) {

        List<SalesData> salesList = salesDataMapper.selectList(
            new LambdaQueryWrapper<SalesData>()
                .eq(SalesData::getSiteCode, siteCode)
                .in(SalesData::getTransactionCategory, List.of("income", "refund", "fee"))
                .ge(SalesData::getTransactionDate, startDate.atStartOfDay())
                .le(SalesData::getTransactionDate, endDate.atTime(23, 59, 59))
        );

        BigDecimal total = BigDecimal.ZERO;

        for (SalesData sales : salesList) {
            BigDecimal fees = sales.getSellingFees().abs()
                .add(sales.getFbaFees().abs())
                .add(sales.getOtherTransactionFees().abs());

            // 获取交易日期汇率
            LocalDate txDate = sales.getTransactionDate().toLocalDate();
            BigDecimal rate = rateService.getRate(txDate, currencyCode);

            total = total.add(fees.multiply(rate));
        }

        return total;
    }

    /**
     * 计算广告费
     */
    private BigDecimal calculateAdvertisingFees(String siteCode, int year, String quarter) {
        // 获取季度月份
        List<String> months = getQuarterMonths(year, quarter);

        List<AdvertisingData> ads = advertisingDataMapper.selectList(
            new LambdaQueryWrapper<AdvertisingData>()
                .eq(AdvertisingData::getSiteCode, siteCode)
                .in(AdvertisingData::getYearMonth, months)
        );

        // 返回已换算的人民币金额汇总
        return ads.stream()
            .map(ad -> {
                if (ad.getAmountCny() != null) {
                    return ad.getAmountCny();
                }
                // 若未预算，现场计算
                BigDecimal rate = rateService.getRate(
                    LocalDate.parse(ad.getYearMonth() + "-01"),
                    ad.getCurrencyCode()
                );
                return ad.getAmount().multiply(rate);
            })
            .reduce(BigDecimal.ZERO, BigDecimal::add);
    }

    /**
     * 统计订单数量
     */
    private int countOrders(String siteCode, LocalDate startDate, LocalDate endDate) {
        return salesDataMapper.countDistinctOrders(siteCode, "income",
            startDate.atStartOfDay(), endDate.atTime(23, 59, 59));
    }

    /**
     * 获取季度日期范围
     */
    private LocalDate[] getQuarterDateRange(int year, String quarter) {
        return switch (quarter) {
            case "Q1" -> new LocalDate[]{
                LocalDate.of(year, 1, 1),
                LocalDate.of(year, 3, 31)
            };
            case "Q2" -> new LocalDate[]{
                LocalDate.of(year, 4, 1),
                LocalDate.of(year, 6, 30)
            };
            case "Q3" -> new LocalDate[]{
                LocalDate.of(year, 7, 1),
                LocalDate.of(year, 9, 30)
            };
            case "Q4" -> new LocalDate[]{
                LocalDate.of(year, 10, 1),
                LocalDate.of(year, 12, 31)
            };
            default -> throw new BusinessException("无效的季度: " + quarter);
        };
    }

    /**
     * 获取季度包含的月份
     */
    private List<String> getQuarterMonths(int year, String quarter) {
        return switch (quarter) {
            case "Q1" -> List.of(year + "-01", year + "-02", year + "-03");
            case "Q2" -> List.of(year + "-04", year + "-05", year + "-06");
            case "Q3" -> List.of(year + "-07", year + "-08", year + "-09");
            case "Q4" -> List.of(year + "-10", year + "-11", year + "-12");
            default -> List.of();
        };
    }

    /**
     * 保存汇总结果到缓存表
     */
    private void saveSummaryCache(SummaryResult result, int year, String quarter) {
        SummaryCache cache = summaryCacheMapper.selectOne(
            new LambdaQueryWrapper<SummaryCache>()
                .eq(SummaryCache::getSummaryYear, year)
                .eq(SummaryCache::getSummaryQuarter, quarter)
                .eq(SummaryCache::getSiteCode, result.getSiteCode())
        );

        if (cache == null) {
            cache = new SummaryCache();
            cache.setSummaryYear(year);
            cache.setSummaryQuarter(quarter);
        }

        cache.setSellerId(result.getMcid());
        cache.setCompanyName(result.getCompanyName());
        cache.setTaxId(result.getTaxId());
        cache.setTotalRevenue(result.getTotalRevenue());
        cache.setRefundAmount(result.getRefundAmount());
        cache.setNetIncome(result.getNetIncome());
        cache.setPlatformFees(result.getPlatformFees());
        cache.setTransactionCount(result.getTransactionCount());
        cache.setCacheStatus("valid");
        cache.setCalculateTime(LocalDateTime.now());
```

### 6.3 登录失败锁定

```java
/**
 * 登录服务
 */
@Service
public class LoginService {

    private static final int MAX_FAIL_COUNT = 5;
    private static final int LOCK_MINUTES = 30;

    @Autowired
    private UserMapper userMapper;

    public LoginResult login(String username, String password) {
        User user = userMapper.selectByUsername(username);

        // 1. 检查账号是否存在
        if (user == null) {
            throw new BusinessException(ErrorCode.LOGIN_FAILED);
        }

        // 2. 检查账号是否锁定
        if (user.getLockTime() != null &&
            user.getLockTime().plusMinutes(LOCK_MINUTES).isAfter(LocalDateTime.now())) {
            throw new BusinessException(ErrorCode.ACCOUNT_LOCKED,
                "账号已锁定，请" + LOCK_MINUTES + "分钟后重试");
        }

        // 3. 验证密码
        if (!passwordUtil.matches(password, user.getPassword())) {
            // 增加失败次数
            int failCount = user.getLoginFailCount() + 1;
            user.setLoginFailCount(failCount);

            if (failCount >= MAX_FAIL_COUNT) {
                user.setLockTime(LocalDateTime.now());
            }

            userMapper.updateById(user);
            throw new BusinessException(ErrorCode.LOGIN_FAILED);
        }

        // 4. 登录成功，重置失败次数
        user.setLoginFailCount(0);
        user.setLockTime(null);
        user.setLastLoginTime(LocalDateTime.now());
        userMapper.updateById(user);

        // 5. 生成Token
        StpUtil.login(user.getId());

        return LoginResult.builder()
            .token(StpUtil.getTokenValue())
            .user(user)
            .build();
    }
}
```

```java
@Service
@Slf4j
public class BatchImportService {

    // 批次大小
    private static final int BATCH_SIZE = 1000;

    @Autowired
    private SqlSessionFactory sqlSessionFactory;

    /**
     * 批量插入销售数据
     */
    @Transactional(rollbackFor = Exception.class)
    public void batchInsertSales(List<SalesData> dataList) {
        if (CollectionUtils.isEmpty(dataList)) {
            return;
        }

        try (SqlSession sqlSession = sqlSessionFactory.openSession(ExecutorType.BATCH)) {
            SalesDataMapper mapper = sqlSession.getMapper(SalesDataMapper.class);

            for (int i = 0; i < dataList.size(); i++) {
                mapper.insert(dataList.get(i));

                // 每BATCH_SIZE条提交一次
                if ((i + 1) % BATCH_SIZE == 0) {
                    sqlSession.flushStatements();
                    sqlSession.clearCache();
                    log.info("批量插入进度: {}/{}", i + 1, dataList.size());
                }
            }

            // 提交剩余数据
            sqlSession.flushStatements();
        }
    }
}
```
