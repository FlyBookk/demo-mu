## 1. 设计概述

### 1.1 背景

基于《字段映射画布交互设计文档》，本文档详细描述字段映射画布功能的后端技术实现方案，包括数据库设计、API接口设计、服务层实现、字段元数据管理等内容。

### 1.2 目标

- 提供目标字段元数据查询接口，支持按数据类型和子类型获取
- 提供文件解析接口（可选，前端也可自行解析）
- 提供智能匹配建议接口
- 优化字段映射模板的存储结构
- 支持销售数据两种格式（原始数据/ERP数据）的差异化处理

### 1.3 技术栈

| 类别 | 技术选型 | 版本 | 用途 |
|------|---------|------|------|
| 语言 | Java | 17+ | 后端开发 |

## 2. 数据库设计

### 2.1 现有表结构回顾

字段映射模板表 `t_field_mapping_template` 已存在，需要进行扩展。

### 2.2 表结构优化

```sql
-- 字段映射模板表（扩展）
ALTER TABLE t_field_mapping_template 
ADD COLUMN sub_type VARCHAR(20) DEFAULT NULL COMMENT '子类型：ORIGINAL-原始数据, ERP-ERP数据（仅销售数据有效）' AFTER data_type,
ADD COLUMN source_fields JSON DEFAULT NULL COMMENT '源字段列表（可选保存）',
ADD COLUMN header_row INT DEFAULT 1 COMMENT '表头行号',
ADD COLUMN default_values JSON DEFAULT NULL COMMENT '默认值配置 [{field, value}]';

-- 更新 mapping_config 字段说明
-- 现有结构：JSON数组 [{source, target}]
-- 扩展后：[{source, target, sourceIndex}]

-- 添加索引
CREATE INDEX idx_data_type_sub ON t_field_mapping_template(data_type, sub_type);
```

### 2.3 字段元数据表（新增）

```sql
-- 目标字段元数据表
CREATE TABLE t_target_field_metadata (
    id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '主键ID',
    data_type VARCHAR(20) NOT NULL COMMENT '数据类型：SALES, SHIPPING, ADVERTISING, RATE',
    sub_type VARCHAR(20) DEFAULT NULL COMMENT '子类型：ORIGINAL, ERP（仅SALES有效）',
    field_name VARCHAR(50) NOT NULL COMMENT '字段名（数据库字段）',
    field_label VARCHAR(100) NOT NULL COMMENT '中文标签',
    field_description VARCHAR(255) DEFAULT NULL COMMENT '字段描述',
    field_type VARCHAR(20) NOT NULL DEFAULT 'string' COMMENT '字段类型：string, number, datetime, boolean',
    required TINYINT NOT NULL DEFAULT 0 COMMENT '是否必填：0-否, 1-是',
    max_length INT DEFAULT NULL COMMENT '最大长度（string类型）',
    precision_value INT DEFAULT NULL COMMENT '精度（number类型）',
    sort_order INT DEFAULT 0 COMMENT '排序序号',
    site_aliases JSON DEFAULT NULL COMMENT '站点字段别名 {US: "order id", DE: "Bestellnummer"}',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    
    UNIQUE KEY uk_type_field (data_type, sub_type, field_name),
    INDEX idx_data_type (data_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='目标字段元数据表';

-- ERP数据聚合规则表
CREATE TABLE t_erp_aggregate_rule (
    id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '主键ID',

    description VARCHAR(100) DEFAULT NULL COMMENT '说明',
    sort_order INT DEFAULT 0 COMMENT '排序',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    
    UNIQUE KEY uk_trans_type (transaction_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='ERP数据聚合规则表';
```

## 5. 服务层实现

### 5.1 字段映射服务

```java
package com.musheng.config.mapping.service;

import com.musheng.config.mapping.entity.*;
import com.musheng.config.mapping.mapper.*;
import com.musheng.config.mapping.dto.*;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.util.*;
import java.util.stream.Collectors;

@Slf4j
@Service
@RequiredArgsConstructor
public class FieldMappingService {
    
    private final TargetFieldMetadataMapper metadataMapper;
    private final ErpAggregateRuleMapper aggregateRuleMapper;
    private final FileParseService fileParseService;
    
    private static final Map<String, String> SUB_TYPE_NAMES = Map.of(
        "ORIGINAL", "亚马逊原始数据",
        "ERP", "ERP结算明细"
    );
    
    /**
     * 获取目标字段定义
     */
    @Cacheable(value = "targetFields", key = "#dataType + ':' + #subType")
    public TargetFieldsResponse getTargetFields(String dataType, String subType) {
        // 查询字段元数据
        List<TargetFieldMetadata> metadataList = metadataMapper.selectByDataType(dataType, subType);
        
        // 转换为VO
        List<TargetFieldVO> fields = metadataList.stream()
            .map(this::convertToVO)
            .collect(Collectors.toList());
        
        // 构建响应
        TargetFieldsResponse response = new TargetFieldsResponse();
        response.setDataType(dataType);
        response.setSubType(subType);
        response.setSubTypeName(SUB_TYPE_NAMES.get(subType));
        response.setFields(fields);
        
        // ERP数据需要返回聚合配置

    /**
     * 智能匹配
     */
    public AutoMatchResponse autoMatch(AutoMatchRequest request) {
        // 获取目标字段
        List<TargetFieldMetadata> targetMetadata = metadataMapper.selectByDataType(
            request.getDataType(), 
            request.getSubType()
        );
        
        List<MatchSuggestion> mappings = new ArrayList<>();
        Set<String> matchedSources = new HashSet<>();
        Set<String> matchedTargets = new HashSet<>();
        
        // 按优先级匹配
        for (String source : request.getSourceFields()) {
            if (matchedSources.contains(source)) continue;
            
            MatchSuggestion suggestion = findBestMatch(
                source, 
                targetMetadata, 
                matchedTargets,
                request.getSiteCode()
            );
            
            if (suggestion != null) {
                mappings.add(suggestion);
                matchedSources.add(source);
                matchedTargets.add(suggestion.getTarget());
            }
        }
        
        // 未匹配的字段
        List<String> unmatchedSource = request.getSourceFields().stream()
            .filter(s -> !matchedSources.contains(s))
            .collect(Collectors.toList());
        
        List<String> unmatchedTarget = targetMetadata.stream()
            .map(TargetFieldMetadata::getFieldName)
            .filter(t -> !matchedTargets.contains(t))
            .collect(Collectors.toList());
        
        AutoMatchResponse response = new AutoMatchResponse();
        response.setMappings(mappings);
        response.setUnmatchedSource(unmatchedSource);
        response.setUnmatchedTarget(unmatchedTarget);
        
        return response;
    }
    
    /**
     * 查找最佳匹配
     */
    private MatchSuggestion findBestMatch(
        String source, 
        List<TargetFieldMetadata> targets,
        Set<String> matchedTargets,
        String siteCode
    ) {
        String normalizedSource = normalize(source);
        
        for (TargetFieldMetadata target : targets) {
            if (matchedTargets.contains(target.getFieldName())) continue;
            
            // 1. 完全匹配
            if (source.equals(target.getFieldName())) {
                return createSuggestion(source, target.getFieldName(), 1.0, "exact");
            }
            
            // 2. 忽略大小写
            if (source.equalsIgnoreCase(target.getFieldName())) {
                return createSuggestion(source, target.getFieldName(), 0.95, "ignore_case");
            }
            
            // 3. 站点别名匹配
            if (siteCode != null && target.getSiteAliases() != null) {
                String alias = target.getSiteAliases().get(siteCode);
                if (alias != null && source.equalsIgnoreCase(alias)) {
                    return createSuggestion(source, target.getFieldName(), 0.9, "alias");
                }
            }
            
            // 4. 中文标签匹配
            if (source.equals(target.getFieldLabel())) {
                return createSuggestion(source, target.getFieldName(), 0.85, "label");
            }
            
            // 5. 规范化匹配
            if (normalizedSource.equals(normalize(target.getFieldName()))) {
                return createSuggestion(source, target.getFieldName(), 0.8, "normalized");
            }
            
            // 6. 包含匹配
            String normalizedTarget = normalize(target.getFieldName());
            if (normalizedSource.contains(normalizedTarget) || normalizedTarget.contains(normalizedSource)) {
                return createSuggestion(source, target.getFieldName(), 0.7, "contains");
            }
        }
        
        // 7. 相似度匹配（Levenshtein距离）
        for (TargetFieldMetadata target : targets) {
            if (matchedTargets.contains(target.getFieldName())) continue;
            
            double similarity = calculateSimilarity(normalizedSource, normalize(target.getFieldName()));
            if (similarity > 0.7) {
                return createSuggestion(source, target.getFieldName(), similarity * 0.8, "similar");
            }
        }
        
        return null;
    }
    
    /**
     * 规范化字符串
     */
    private String normalize(String str) {
        return str.toLowerCase().replaceAll("[\\-_\\s.]", "");
    }
    
    /**
     * 计算相似度（Levenshtein距离）
     */
    private double calculateSimilarity(String s1, String s2) {
        int distance = levenshteinDistance(s1, s2);
        int maxLength = Math.max(s1.length(), s2.length());
        return 1.0 - (double) distance / maxLength;
    }
    
    private int levenshteinDistance(String s1, String s2) {
        int[][] dp = new int[s1.length() + 1][s2.length() + 1];
        
        for (int i = 0; i <= s1.length(); i++) dp[i][0] = i;
        for (int j = 0; j <= s2.length(); j++) dp[0][j] = j;
        
        for (int i = 1; i <= s1.length(); i++) {
            for (int j = 1; j <= s2.length(); j++) {
                int cost = s1.charAt(i - 1) == s2.charAt(j - 1) ? 0 : 1;
                dp[i][j] = Math.min(
                    Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1),
                    dp[i - 1][j - 1] + cost
                );
            }
        }
        
        return dp[s1.length()][s2.length()];
    }
    
    private MatchSuggestion createSuggestion(String source, String target, double confidence, String matchType) {
        MatchSuggestion suggestion = new MatchSuggestion();
        suggestion.setSource(source);
        suggestion.setTarget(target);
        suggestion.setConfidence(confidence);
        suggestion.setMatchType(matchType);
        return suggestion;
    }
    
    private TargetFieldVO convertToVO(TargetFieldMetadata metadata) {
        TargetFieldVO vo = new TargetFieldVO();
        vo.setField(metadata.getFieldName());
        vo.setLabel(metadata.getFieldLabel());
        vo.setDescription(metadata.getFieldDescription());
        vo.setType(metadata.getFieldType());
        vo.setRequired(metadata.getRequired());
        vo.setMaxLength(metadata.getMaxLength());
        vo.setPrecision(metadata.getPrecision());
        return vo;
    }
}
```

```java
package com.musheng.config.mapping.service;

import com.alibaba.excel.context.AnalysisContext;

import com.musheng.config.mapping.dto.*;

@Service
public class FileParseService {
    
    private static final int MAX_PREVIEW_ROWS = 20;
    private static final int MAX_SAMPLE_LENGTH = 30;
    
    /**
     * 预览文件
     */
    public FilePreviewResponse previewFile(MultipartFile file, Integer previewRows) {
        String fileName = file.getOriginalFilename();
        previewRows = Math.min(previewRows, MAX_PREVIEW_ROWS);
        
        try {
            if (fileName != null && fileName.endsWith(".csv")) {

            } else {
                return previewExcelFile(file, previewRows);

    /**
     * 预览CSV文件
     */
    private FilePreviewResponse previewCsvFile(MultipartFile file, int previewRows) throws IOException {
        Charset charset = detectEncoding(file);
        char delimiter = detectCsvDelimiter(file, charset);
        
        List<FilePreviewRow> rows = new ArrayList<>();
        int totalRows = 0;
        
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(file.getInputStream(), charset))) {
            
            CSVFormat format = CSVFormat.DEFAULT.withDelimiter(delimiter);
            CSVParser parser = CSVParser.parse(reader, format);
            
            for (CSVRecord record : parser) {
                totalRows++;
                if (totalRows <= previewRows) {
                    List<String> cells = new ArrayList<>();
                    record.forEach(cells::add);
                    
                    FilePreviewRow row = new FilePreviewRow();
                    row.setRowNum(totalRows);
                    row.setContent(String.join("\t", cells));
                    row.setCells(cells);
                    rows.add(row);
                }
            }
        }
        
        FilePreviewResponse response = new FilePreviewResponse();
        response.setRows(rows);
        response.setTotalRows(totalRows);
        response.setEncoding(charset.name());
        response.setDelimiter(String.valueOf(delimiter));
        
        return response;
    }

     * 预览Excel文件

        List<String> sheets = new ArrayList<>();

    private ParseFieldsResponse parseCsvFields(MultipartFile file, int headerRow) throws IOException {

        List<String> headerFields = null;
        List<String> sampleValues = null;
        int currentRow = 0;

        response.setTotalColumns(headerFields.size());

        response.setEncoding(charset.name());
        response.setDelimiter(String.valueOf(delimiter));

    private ParseFieldsResponse parseExcelFields(MultipartFile file, int headerRow, String sheetName)

        List<SourceFieldVO> fields = new ArrayList<>();
        int[] totalColumns = {0};

                                row.add(data.getOrDefault(i, ""));
                            }

                            totalColumns[0] = Math.max(totalColumns[0], data.size());
                        }
                    }
                    
                    @Override
                    public void doAfterAllAnalysed(AnalysisContext context) {}
                })
                .sheet(sheetName != null ? sheetName : null)
                .doRead();
            
            if (rows.isEmpty()) {
                throw new BusinessException("未找到表头行");
            }
            
            List<String> headerFields = rows.get(0);
            List<String> sampleValues = rows.size() > 1 ? rows.get(1) : null;
            
            for (int i = 0; i < headerFields.size(); i++) {
                String name = headerFields.get(i).trim();
                if (name.isEmpty()) continue;
                
                SourceFieldVO field = new SourceFieldVO();
                field.setName(name);
                field.setIndex(i);
                
                if (sampleValues != null && i < sampleValues.size()) {
                    String sample = sampleValues.get(i);
                    field.setSample(sample.length() > MAX_SAMPLE_LENGTH 
                        ? sample.substring(0, MAX_SAMPLE_LENGTH) + "..." 
                        : sample);
                }
                
                fields.add(field);
            }
        }
        
        ParseFieldsResponse response = new ParseFieldsResponse();
        response.setFields(fields);
        response.setTotalColumns(totalColumns[0]);
        response.setHeaderRow(headerRow);
        response.setEncoding("UTF-8");
        response.setDelimiter("\t");

    /**
     * 检测文件编码
     */
    private Charset detectEncoding(MultipartFile file) throws IOException {
        byte[] bytes = new byte[1024];
        try (InputStream is = file.getInputStream()) {
            int len = is.read(bytes);
            
            // 检测 BOM
            if (len >= 3 && bytes[0] == (byte) 0xEF && bytes[1] == (byte) 0xBB && bytes[2] == (byte) 0xBF) {
                return StandardCharsets.UTF_8;
            }
            
            // 简单检测 GBK（中文字符）
            boolean hasHighByte = false;
            for (int i = 0; i < len; i++) {
                if ((bytes[i] & 0x80) != 0) {
                    hasHighByte = true;
                    break;
                }
            }
            
            if (hasHighByte) {
                // 尝试 UTF-8 解码
                try {
                    new String(bytes, 0, len, StandardCharsets.UTF_8);
                    return StandardCharsets.UTF_8;
                } catch (Exception e) {
                    return Charset.forName("GBK");
                }
            }
            
            return StandardCharsets.UTF_8;
        }
    }
    
    /**
     * 检测CSV分隔符
     */
    private char detectCsvDelimiter(MultipartFile file, Charset charset) throws IOException {
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(file.getInputStream(), charset))) {
            
            String firstLine = reader.readLine();
            if (firstLine == null) return ',';
            
            // 统计分隔符出现次数
            int tabCount = countChar(firstLine, '\t');
            int commaCount = countChar(firstLine, ',');
            int semicolonCount = countChar(firstLine, ';');
            
            if (tabCount > commaCount && tabCount > semicolonCount) return '\t';
            if (semicolonCount > commaCount) return ';';
            return ',';
        }
    }
    
    private int countChar(String str, char c) {
        int count = 0;
        for (char ch : str.toCharArray()) {
            if (ch == c) count++;
        }
        return count;
    }
```

## 10. 技术决策说明

### 10.1 文件解析位置

**决策**：前后端都支持，推荐前端解析

**理由**：
- 前端解析：响应快，无需上传，用户体验好
- 后端解析：编码检测更准确，大文件处理更稳定
- 提供后端接口作为备选方案，前端可按需调用

### 10.2 字段元数据存储

**决策**：使用数据库表 + 缓存

**理由**：
- 便于动态维护和扩展
- 支持站点别名配置
- 通过缓存保证查询性能

### 10.3 智能匹配位置

**决策**：前后端都实现

**理由**：
- 前端实现：响应快，无网络延迟
- 后端实现：可利用站点别名等元数据，匹配更准确
- 前端可先尝试本地匹配，效果不佳时调用后端
