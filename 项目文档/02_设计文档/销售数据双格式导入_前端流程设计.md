## 一、功能概述

实现销售数据的双格式导入功能，支持：
1. **亚马逊原始报告** - 直接导入
2. **ERP结算明细** - 按订单合并后导入

采用模板化字段映射方案，通过预置模板配置支持不同格式的数据导入。

### 2.2 步骤详细设计

#### 步骤1: 选择站点

| 项目 | 说明 |
|------|------|
| **交互** | 下拉选择框 |
| **数据源** | 调用 `getEnabledMarketplaces()` 获取站点列表 |
| **校验** | 必选 |
| **联动** | 选择后重置步骤2、3的选择 |

#### 步骤2: 选择数据类型

| 项目 | 说明 |
|------|------|
| **交互** | 单选按钮组 (Radio Group) |
| **选项** | 亚马逊原始报告 / ERP结算明细 |
| **校验** | 必选 |
| **联动** | 选择后重置步骤3的模板选择，自动选中默认模板 |
| **提示** | 根据类型显示不同说明文字 |

#### 步骤3: 选择映射模板

| 项目 | 说明 |
|------|------|
| **交互** | 下拉选择框 + 模板预览折叠面板 |
| **数据源** | 根据站点+数据类型筛选模板列表 |
| **校验** | 必选；无可用模板时提示前往配置 |
| **联动** | 选择后可展开查看映射关系 |

#### 步骤4: 上传文件

| 项目 | 说明 |
|------|------|
| **交互** | 拖拽上传区域 |
| **格式** | CSV, XLSX, XLS |
| **限制** | 单文件最大 50MB |
| **校验** | 文件类型、大小、编码检测 |
| **反馈** | 上传进度、成功/失败状态 |

#### 步骤5: 预览确认

| 项目 | 说明 |
|------|------|
| **配置摘要** | 展示已选的站点、数据类型、模板、文件信息 |
| **数据预览** | 展示前10行解析后的数据 |
| **统计信息** | 总行数、预计导入数、ERP合并数 |
| **操作** | 上一步、确认导入 |

### 3.3 组合式函数 useSalesImport.ts

```typescript
// composables/useSalesImport.ts
import { ref, reactive } from 'vue'
import { message } from 'ant-design-vue'
import type { Marketplace } from '@/types/marketplace'
import type { FieldMappingTemplate } from '@/types/fieldMapping'
import type { SalesSourceType, SalesImportParams, SalesPreviewData } from '@/types/sales'
import type { ImportRecord } from '@/types/importRecord'
import { getEnabledMarketplaces } from '@/api/marketplace'
import { getTemplatesBySiteAndType } from '@/api/fieldMapping'
import { uploadSalesFile, previewSalesImport, importSalesData } from '@/api/sales'

export function useSalesImport() {
  // 站点列表
  const marketplaces = ref<Marketplace[]>([])
  
  // 模板列表
  const templates = ref<FieldMappingTemplate[]>([])
  
  // 导入表单
  const importForm = reactive<{
    siteCode: string | undefined
    sourceType: SalesSourceType | undefined
    templateId: number | undefined
    quarter?: string
  }>({
    siteCode: undefined,
    sourceType: undefined,
    templateId: undefined,
    quarter: undefined
  })
  
  // 上传文件信息
  const uploadedFile = ref<{
    fileId: string
    fileName: string
    fileSize: number
  } | null>(null)
  
  // 预览数据
  const previewData = ref<SalesPreviewData[]>([])
  const previewStats = ref({
    totalRows: 0,
    mergedRows: 0,
    errorRows: 0
  })
  
  // 导入状态
  const importing = ref(false)
  const importResult = ref<ImportRecord | null>(null)
  const showResultModal = ref(false)
  
  // 获取站点列表
  async function fetchMarketplaces() {
    try {
      const res = await getEnabledMarketplaces()
      marketplaces.value = res.data || []
    } catch (error) {
      console.error('获取站点列表失败:', error)
      message.error('获取站点列表失败')
    }
  }
  
  // 获取模板列表
  async function fetchTemplates() {
    try {
      const res = await getTemplatesBySiteAndType({
        dataType: 'sales'
      })
      templates.value = res.data || []
    } catch (error) {
      console.error('获取模板列表失败:', error)
      message.error('获取模板列表失败')
    }
  }
  
  // 上传文件
  async function uploadFile(file: File): Promise<boolean> {
    try {
      const res = await uploadSalesFile(file, importForm.siteCode!, importForm.sourceType!)
      uploadedFile.value = {
        fileId: res.data.fileId,
        fileName: res.data.fileName,
        fileSize: res.data.fileSize
      }
      message.success('文件上传成功')
      return true
    } catch (error) {
      console.error('文件上传失败:', error)
      message.error('文件上传失败')
      return false
    }
  }
  
  // 预览导入数据
  async function previewImport() {
    if (!uploadedFile.value) return
    
    try {
      const res = await previewSalesImport({
        siteCode: importForm.siteCode!,
        sourceType: importForm.sourceType!,
        templateId: importForm.templateId!,
        fileId: uploadedFile.value.fileId
      })
      
      previewData.value = res.data.records || []
      previewStats.value = {
        totalRows: res.data.totalRows,
        mergedRows: res.data.mergedRows,
        errorRows: res.data.errorRows
      }
    } catch (error) {
      console.error('预览数据失败:', error)
      message.error('预览数据失败')
    }
  }
  
  // 开始导入
  async function startImport() {
    if (!uploadedFile.value) return
    
    importing.value = true
    try {
      const params: SalesImportParams = {
        siteCode: importForm.siteCode!,
        sourceType: importForm.sourceType!,
        templateId: importForm.templateId!,
        fileId: uploadedFile.value.fileId,
        quarter: importForm.quarter
      }
      
      const res = await importSalesData(params)
      importResult.value = res.data
      showResultModal.value = true
      message.success('导入任务已提交')
    } catch (error) {
      console.error('导入失败:', error)
      message.error('导入失败')
    } finally {
      importing.value = false
    }
  }
  
  // 重置表单
  function resetForm() {
    importForm.siteCode = undefined
    importForm.sourceType = undefined
    importForm.templateId = undefined
    importForm.quarter = undefined
    uploadedFile.value = null
    previewData.value = []
    previewStats.value = { totalRows: 0, mergedRows: 0, errorRows: 0 }
    importResult.value = null
    showResultModal.value = false
  }
  
  return {
    marketplaces,
    templates,
    importForm,
    uploadedFile,
    previewData,
    previewStats,
    importing,
    importResult,
    showResultModal,
    fetchMarketplaces,
    fetchTemplates,
    uploadFile,
    previewImport,
    startImport,
    resetForm
  }
}
```

### 3.4 模板筛选逻辑 useTemplateFilter.ts

```typescript
// composables/useTemplateFilter.ts
import { computed, type Ref, type ComputedRef } from 'vue'
import type { FieldMappingTemplate } from '@/types/fieldMapping'
import type { SalesSourceType } from '@/types/sales'

interface ImportForm {
  siteCode: string | undefined
  sourceType: SalesSourceType | undefined
  templateId: number | undefined
}

export function useTemplateFilter(
  importForm: ImportForm,
  templates: Ref<FieldMappingTemplate[]>
) {
  // 根据站点和数据类型筛选模板
  const filteredTemplates: ComputedRef<FieldMappingTemplate[]> = computed(() => {
    if (!importForm.siteCode || !importForm.sourceType) {
      return []
    }
    
    return templates.value.filter(t => 
      t.siteCode === importForm.siteCode &&
      t.sourceType === importForm.sourceType &&
      t.dataType === 'sales' &&
      t.status === 1
    )
  })
  
  // 获取当前选中的模板
  const selectedTemplate: ComputedRef<FieldMappingTemplate | undefined> = computed(() => {
    if (!importForm.templateId) return undefined
    return templates.value.find(t => t.id === importForm.templateId)
  })
  
  return {
    filteredTemplates,
    selectedTemplate
  }
}
```

```typescript
import { request } from '@/utils/request'
import type { 
  SalesImportParams, 
  SalesPreviewResult, 
  SalesSourceType 
} from '@/types/sales'
import type { ImportRecord } from '@/types/importRecord'

const BASE_URL = '/api/v1/business/sales'

/**
 * 上传销售数据文件
 */
export function uploadSalesFile(
  file: File, 
  siteCode: string, 
  sourceType: SalesSourceType
) {
  const formData = new FormData()
  formData.append('file', file)
  formData.append('siteCode', siteCode)
  formData.append('sourceType', sourceType)
  
  return request.post<{
    fileId: string
    fileName: string
    fileSize: number
    headers: string[]
  }>(`${BASE_URL}/upload`, formData, {
    headers: { 'Content-Type': 'multipart/form-data' }
  })
}

export function previewSalesImport(params: {
  siteCode: string
  sourceType: SalesSourceType
  templateId: number
  fileId: string

  return request.post<SalesPreviewResult>(`${BASE_URL}/preview`, params)
}

 * 执行销售数据导入
 */
export function importSalesData(params: SalesImportParams) {
  return request.post<ImportRecord>(`${BASE_URL}/import`, params)
```

## 六、状态管理

### 6.1 Pinia Store (可选)

如果导入状态需要跨组件共享，可以使用 Pinia Store：

```typescript
// stores/salesImport.ts
import { defineStore } from 'pinia'
import type { SalesSourceType } from '@/types/sales'
import type { FieldMappingTemplate } from '@/types/fieldMapping'

export const useSalesImportStore = defineStore('salesImport', {
  state: () => ({
    // 当前步骤
    currentStep: 0,
    
    // 表单数据
    siteCode: undefined as string | undefined,
    sourceType: undefined as SalesSourceType | undefined,
    templateId: undefined as number | undefined,
    
    // 模板列表
    templates: [] as FieldMappingTemplate[],
    
    // 上传文件
    uploadedFile: null as { fileId: string; fileName: string; fileSize: number } | null,
    
    // 预览数据
    previewData: [],
    previewStats: { totalRows: 0, mergedRows: 0, errorRows: 0 },
    
    // 状态
    loading: false,
    importing: false
  }),
  
  getters: {
    filteredTemplates: (state) => {
      if (!state.siteCode || !state.sourceType) return []
      return state.templates.filter(t => 
        t.siteCode === state.siteCode && 
        t.sourceType === state.sourceType
      )
    },
    
    selectedTemplate: (state) => {
      return state.templates.find(t => t.id === state.templateId)
    },
    
    canProceed: (state) => {
      switch (state.currentStep) {
        case 0: return !!state.siteCode
        case 1: return !!state.sourceType
        case 2: return !!state.templateId
        case 3: return !!state.uploadedFile
        default: return true
      }
    }
  },
  
  actions: {
    nextStep() {
      if (this.canProceed && this.currentStep < 4) {
        this.currentStep++
      }
    },
    
    prevStep() {
      if (this.currentStep > 0) {
        this.currentStep--
      }
    },
    
    reset() {
      this.currentStep = 0
      this.siteCode = undefined
      this.sourceType = undefined
      this.templateId = undefined
      this.uploadedFile = null
      this.previewData = []
      this.previewStats = { totalRows: 0, mergedRows: 0, errorRows: 0 }
    }
  }
})
```

### 8.1 文件上传错误

```typescript
// 文件上传前校验
function beforeUpload(file: File): boolean | Promise<boolean> {
  // 1. 文件类型校验
  const isValidType = [
    'text/csv',
    'application/vnd.ms-excel',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
  ].includes(file.type) || file.name.endsWith('.csv')
  
  if (!isValidType) {
    message.error('只支持 CSV 或 Excel 文件')
    return false
  }
  
  // 2. 文件大小校验
  const maxSize = 50 * 1024 * 1024 // 50MB
  if (file.size > maxSize) {
    message.error('文件大小不能超过 50MB')
    return false
  }
  
  return true
}
```
