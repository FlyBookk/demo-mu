## 一、功能概述

实现销售数据的双格式导入后端服务，支持：
1. **亚马逊原始报告** - 直接字段映射解析
2. **ERP结算明细** - 行转列聚合后导入

采用模板化字段映射 + 策略模式，通过解析器工厂实现不同格式的数据解析。

```java
import lombok.Getter;

/**
 * 销售数据来源类型

@Getter
public enum SalesSourceType {
    
    AMAZON_ORIGINAL("AMAZON_ORIGINAL", "亚马逊原始报告"),
    ERP_SETTLEMENT("ERP_SETTLEMENT", "ERP结算明细");
    
    private final String code;
    private final String description;
    
    SalesSourceType(String code, String description) {
        this.code = code;
        this.description = description;
    }
    
    public static SalesSourceType fromCode(String code) {
        for (SalesSourceType type : values()) {
            if (type.getCode().equals(code)) {
                return type;
            }
        }
        throw new IllegalArgumentException("Unknown source type: " + code);
    }
}
```

### 4.6 亚马逊原始数据解析器

```java
package com.musheng.business.sales.parser;

import com.musheng.business.sales.converter.DateConverter;
import com.musheng.business.sales.converter.NumberConverter;
import com.musheng.business.sales.converter.TransactionTypeMapper;
import com.musheng.business.sales.dto.ParseContext;
import com.musheng.business.sales.dto.ParseResult;
import com.musheng.business.sales.entity.SalesData;
import com.musheng.common.enums.SalesSourceType;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVParser;
import org.apache.commons.csv.CSVRecord;
import org.springframework.stereotype.Component;
import org.springframework.web.multipart.MultipartFile;

import java.io.*;
import java.math.BigDecimal;
import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.*;

/**
 * 亚马逊原始数据解析器
 * 处理 CustomUnifiedTransaction / CustomTransaction 格式
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class AmazonOriginalParser implements SalesDataParser {
    
    private final DateConverter dateConverter;
    private final NumberConverter numberConverter;
    private final TransactionTypeMapper transactionTypeMapper;
    
    // 跳过的说明行数（亚马逊报告前7-8行是说明文字）
    private static final int SKIP_ROWS = 8;
    
    @Override
    public List<String> parseHeaders(MultipartFile file) {
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(file.getInputStream(), StandardCharsets.UTF_8))) {
            
            // 跳过说明行，找到真正的表头
            String line;
            int lineNum = 0;
            while ((line = reader.readLine()) != null) {
                lineNum++;
                // 检测是否是表头行（包含关键字段）
                if (isHeaderRow(line)) {
                    return parseHeaderLine(line);
                }
                if (lineNum > 15) {
                    break; // 最多检查15行
                }
            }
            
            throw new RuntimeException("无法识别文件表头");
        } catch (IOException e) {
            throw new RuntimeException("读取文件失败: " + e.getMessage(), e);
        }
    }
    
    @Override
    public ParseResult preview(ParseContext context, int limit) {
        return parseInternal(context, limit);
    }
    
    @Override
    public ParseResult parse(ParseContext context) {
        return parseInternal(context, -1); // -1 表示解析全部
    }
    
    private ParseResult parseInternal(ParseContext context, int limit) {
        ParseResult result = new ParseResult();
        Map<String, String> fieldMapping = context.getFieldMapping();
        
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(context.getInputStream(), StandardCharsets.UTF_8))) {
            
            // 1. 跳过说明行，定位表头
            String line;
            List<String> headers = null;
            int lineNum = 0;
            
            while ((line = reader.readLine()) != null) {
                lineNum++;
                if (isHeaderRow(line)) {
                    headers = parseHeaderLine(line);
                    break;
                }
            }
            
            if (headers == null) {
                result.addError(0, "无法识别文件表头");
                return result;
            }
            
            // 2. 检测语言区域（英文/德文）
            String region = detectRegion(headers);
            Map<String, String> headerMapping = getHeaderMapping(region);
            
            // 3. 解析数据行
            CSVParser parser = CSVFormat.DEFAULT.parse(reader);
            int dataRowNum = 0;
            
            for (CSVRecord record : parser) {
                dataRowNum++;
                result.setTotalRows(result.getTotalRows() + 1);
                
                // 限制预览行数
                if (limit > 0 && result.getSuccessCount() >= limit) {
                    break;
                }
                
                try {
                    SalesData salesData = parseRecord(record, headers, headerMapping, 
                            fieldMapping, context, dataRowNum);
                    
                    if (salesData != null) {
                        result.addData(salesData);
                    }
                } catch (Exception e) {
                    result.addError(dataRowNum, e.getMessage());
                    log.warn("解析第{}行失败: {}", dataRowNum, e.getMessage());
                }
            }
            
            result.setMergedRows(result.getSuccessCount()); // 原始数据不需要合并
            
        } catch (IOException e) {
            result.addError(0, "读取文件失败: " + e.getMessage());
        }
        
        return result;
    }
    
    /**
     * 解析单行数据
     */
    private SalesData parseRecord(CSVRecord record, List<String> headers,
                                   Map<String, String> headerMapping,
                                   Map<String, String> fieldMapping,
                                   ParseContext context, int rowNum) {
        
        // 构建行数据Map
        Map<String, String> rowData = new HashMap<>();
        for (int i = 0; i < Math.min(headers.size(), record.size()); i++) {
            String header = headers.get(i).toLowerCase().trim();
            // 先用原始表头映射到标准表头
            String standardField = headerMapping.getOrDefault(header, header);
            rowData.put(standardField, record.get(i).trim());
        }
        
        // 检查订单号是否存在
        String orderId = getMappedValue(rowData, fieldMapping, "orderId");
        if (orderId == null || orderId.isEmpty()) {
            return null; // 跳过空行
        }
        
        SalesData data = new SalesData();
        data.setImportBatchId(context.getImportBatchId());
        data.setSourceType(SalesSourceType.AMAZON_ORIGINAL.getCode());
        data.setOrderId(orderId);
        
        // 解析站点（从marketplace字段自动识别）
        String marketplace = getMappedValue(rowData, fieldMapping, "marketplace");
        String siteCode = MarketplaceMapper.getSiteCode(marketplace);
        data.setSiteCode(siteCode != null ? siteCode : context.getSiteCode());
        data.setMarketplace(marketplace);
        
        // 解析日期
        String dateStr = getMappedValue(rowData, fieldMapping, "transactionDate");
        if (dateStr != null && !dateStr.isEmpty()) {
            data.setOriginalDateStr(dateStr);
            LocalDateTime dateTime = dateConverter.convert(dateStr, data.getSiteCode());
            data.setTransactionDate(dateTime);
        }
        
        // 解析交易类型
        String transactionType = getMappedValue(rowData, fieldMapping, "transactionType");
        data.setTransactionType(transactionType);
        data.setTransactionCategory(transactionTypeMapper.toCategory(transactionType));
        
        // 解析基础字段
        data.setSettlementId(getMappedValue(rowData, fieldMapping, "settlementId"));
        data.setSku(getMappedValue(rowData, fieldMapping, "sku"));
        data.setDescription(getMappedValue(rowData, fieldMapping, "description"));
        data.setFulfillment(getMappedValue(rowData, fieldMapping, "fulfillment"));
        data.setOrderCity(getMappedValue(rowData, fieldMapping, "orderCity"));
        data.setOrderState(getMappedValue(rowData, fieldMapping, "orderState"));
        data.setOrderPostal(getMappedValue(rowData, fieldMapping, "orderPostal"));
        data.setTaxCollectionModel(getMappedValue(rowData, fieldMapping, "taxCollectionModel"));
        
        // 解析数量
        String quantityStr = getMappedValue(rowData, fieldMapping, "quantity");
        if (quantityStr != null && !quantityStr.isEmpty()) {
            data.setQuantity(Integer.parseInt(quantityStr.replace(",", "")));
        }
        
        // 解析金额字段
        data.setProductSales(parseDecimal(rowData, fieldMapping, "productSales", data.getSiteCode()));
        data.setProductSalesTax(parseDecimal(rowData, fieldMapping, "productSalesTax", data.getSiteCode()));
        data.setShippingCredits(parseDecimal(rowData, fieldMapping, "shippingCredits", data.getSiteCode()));
        data.setShippingCreditsTax(parseDecimal(rowData, fieldMapping, "shippingCreditsTax", data.getSiteCode()));
        data.setGiftWrapCredits(parseDecimal(rowData, fieldMapping, "giftWrapCredits", data.getSiteCode()));
        data.setGiftWrapCreditsTax(parseDecimal(rowData, fieldMapping, "giftWrapCreditsTax", data.getSiteCode()));
        data.setRegulatoryFee(parseDecimal(rowData, fieldMapping, "regulatoryFee", data.getSiteCode()));
        data.setRegulatoryFeeTax(parseDecimal(rowData, fieldMapping, "regulatoryFeeTax", data.getSiteCode()));
        data.setPromotionalRebates(parseDecimal(rowData, fieldMapping, "promotionalRebates", data.getSiteCode()));
        data.setPromotionalRebatesTax(parseDecimal(rowData, fieldMapping, "promotionalRebatesTax", data.getSiteCode()));
        data.setMarketplaceWithheldTax(parseDecimal(rowData, fieldMapping, "marketplaceWithheldTax", data.getSiteCode()));
        data.setSellingFees(parseDecimal(rowData, fieldMapping, "sellingFees", data.getSiteCode()));
        data.setFbaFees(parseDecimal(rowData, fieldMapping, "fbaFees", data.getSiteCode()));
        data.setOtherTransactionFees(parseDecimal(rowData, fieldMapping, "otherTransactionFees", data.getSiteCode()));
        data.setOther(parseDecimal(rowData, fieldMapping, "other", data.getSiteCode()));
        data.setTotal(parseDecimal(rowData, fieldMapping, "total", data.getSiteCode()));
        
        return data;
    }
    
    /**
     * 检测是否为表头行
     */
    private boolean isHeaderRow(String line) {
        String lower = line.toLowerCase();
        return lower.contains("date/time") || 
               lower.contains("datum/uhrzeit") ||
               lower.contains("order id") ||
               lower.contains("bestellnummer");
    }
    
    /**
     * 解析表头行
     */
    private List<String> parseHeaderLine(String line) {
        // 移除BOM和引号
        line = line.replace("\uFEFF", "").replace("\"", "");
        return Arrays.asList(line.split(","));
    }
    
    /**
     * 检测语言区域
     */
    private String detectRegion(List<String> headers) {
        String firstHeader = headers.get(0).toLowerCase().trim();
        if (firstHeader.contains("datum")) {
            return "DE";
        }
        for (String h : headers) {
            if (h.toLowerCase().contains("postage credits")) {
                return "UK";
            }
        }
        return "US";
    }
    
    /**
     * 获取区域对应的表头映射
     */
    private Map<String, String> getHeaderMapping(String region) {
        return AmazonHeaderMapping.getMapping(region);
    }
    
    /**
     * 获取映射后的字段值
     */
    private String getMappedValue(Map<String, String> rowData, 
                                   Map<String, String> fieldMapping, 
                                   String targetField) {
        // 先从模板映射获取源字段名
        String sourceField = fieldMapping.get(targetField);
        if (sourceField != null) {
            String value = rowData.get(sourceField.toLowerCase());
            if (value != null) return value;
        }
        // 尝试直接匹配
        return rowData.get(targetField.toLowerCase());
    }
    
    /**
     * 解析金额字段
     */
    private BigDecimal parseDecimal(Map<String, String> rowData, 
                                     Map<String, String> fieldMapping,
                                     String targetField, String siteCode) {
        String value = getMappedValue(rowData, fieldMapping, targetField);
        if (value == null || value.isEmpty()) {
            return BigDecimal.ZERO;
        }
        return numberConverter.convert(value, siteCode);
    }
}
```

```java
package com.musheng.business.sales.parser;

import com.musheng.business.sales.dto.ErpSettlementRow;
import com.musheng.business.sales.dto.ParseContext;
import com.musheng.business.sales.dto.ParseResult;
import com.musheng.business.sales.entity.SalesData;
import com.musheng.common.enums.SalesSourceType;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVParser;
import org.apache.commons.csv.CSVRecord;
import org.springframework.stereotype.Component;
import org.springframework.web.multipart.MultipartFile;

import java.io.*;
import java.math.BigDecimal;
import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.*;
import java.util.stream.Collectors;

/**
 * ERP结算数据解析器
 * 处理结算中心导出的明细数据，需要行转列聚合
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class ErpSettlementParser implements SalesDataParser {
    
    // 中文表头到标准字段的映射
    private static final Map<String, String> HEADER_MAPPING = Map.ofEntries(
        Map.entry("结算编号", "settlementId"),
        Map.entry("订单号", "orderId"),
        Map.entry("店铺", "storeName"),
        Map.entry("国家", "siteCode"),
        Map.entry("报告类型", "reportType"),
        Map.entry("配送方式", "fulfillment"),
        Map.entry("来源", "source"),
        Map.entry("MSKU", "msku"),
        Map.entry("交易类型", "transactionType"),
        Map.entry("结算时间", "transactionDate"),
        Map.entry("币种", "currencyCode"),
        Map.entry("金额", "amount"),
        Map.entry("数量", "quantity"),
        Map.entry("结算状态", "settlementStatus"),
        Map.entry("转账状态", "transferStatus"),
        Map.entry("Settlement ID", "amazonSettlementId"),
        Map.entry("SKU", "sku"),
        Map.entry("品名", "description"),
        Map.entry("FNSKU", "fnsku")
    );
    
    // 交易类型到金额字段的映射
    private static final Map<String, String> TYPE_TO_FIELD = Map.ofEntries(
        // 收入类
        Map.entry("Principal", "productSales"),
        Map.entry("Tax", "productSalesTax"),
        Map.entry("ShippingCharge", "shippingCredits"),
        Map.entry("ShippingTax", "shippingCreditsTax"),
        Map.entry("GiftWrapCharge", "giftWrapCredits"),
        Map.entry("GiftWrapTax", "giftWrapCreditsTax"),
        // 促销类
        Map.entry("PromotionDiscount", "promotionalRebates"),
        Map.entry("PromotionDiscountTax", "promotionalRebatesTax"),
        // 费用类
        Map.entry("Commission", "sellingFees"),
        Map.entry("FBAPerUnitFulfillmentFee", "fbaFees"),
        Map.entry("DigitalServicesFee", "otherTransactionFees"),
        Map.entry("RefundCommission", "sellingFees"),
        // 税费类
        Map.entry("MarketplaceFacilitatorVAT-Principal", "marketplaceWithheldTax"),
        Map.entry("MarketplaceFacilitatorTax-Principal", "marketplaceWithheldTax"),
        // 其他
        Map.entry("Revenue", "other"),
        Map.entry("FeeAmount", "other")
    );
    
    // 来源到交易分类的映射
    private static final Map<String, String> SOURCE_TO_CATEGORY = Map.of(
        "Shipment", "income",
        "Refund", "refund",
        "RemovalShipment", "adjustment",
        "REVERSAL", "adjustment"
    );
    
    @Override
    public List<String> parseHeaders(MultipartFile file) {
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(file.getInputStream(), StandardCharsets.UTF_8))) {
            
            String headerLine = reader.readLine();
            if (headerLine != null) {
                return Arrays.asList(headerLine.split(","));
            }
            throw new RuntimeException("文件为空");
        } catch (IOException e) {
            throw new RuntimeException("读取文件失败: " + e.getMessage(), e);
        }
    }
    
    @Override
    public ParseResult preview(ParseContext context, int limit) {
        return parseInternal(context, limit);
    }
    
    @Override
    public ParseResult parse(ParseContext context) {
        return parseInternal(context, -1);
    }
    
    private ParseResult parseInternal(ParseContext context, int limit) {
        ParseResult result = new ParseResult();
        
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(context.getInputStream(), StandardCharsets.UTF_8))) {
            
            // 1. 解析表头
            String headerLine = reader.readLine();
            List<String> headers = parseHeaders(headerLine);
            Map<Integer, String> headerIndexMap = buildHeaderIndexMap(headers);
            
            // 2. 解析所有明细行
            List<ErpSettlementRow> allRows = new ArrayList<>();
            CSVParser parser = CSVFormat.DEFAULT.parse(reader);
            int rowNum = 0;
            
            for (CSVRecord record : parser) {
                rowNum++;
                result.setTotalRows(result.getTotalRows() + 1);
                
                try {
                    ErpSettlementRow row = parseRow(record, headerIndexMap, rowNum);
                    if (row != null) {
                        allRows.add(row);
                    }
                } catch (Exception e) {
                    result.addError(rowNum, e.getMessage());
                }
            }
            
            // 3. 按 (订单号 + 站点 + SKU) 分组并聚合
            Map<String, List<ErpSettlementRow>> grouped = allRows.stream()
                .collect(Collectors.groupingBy(r -> 
                    r.getOrderId() + "|" + r.getSiteCode() + "|" + r.getSku()));
            
            // 4. 聚合生成 SalesData
            int aggregatedCount = 0;
            for (Map.Entry<String, List<ErpSettlementRow>> entry : grouped.entrySet()) {
                if (limit > 0 && aggregatedCount >= limit) {
                    break;
                }
                
                try {
                    SalesData salesData = aggregateRows(entry.getValue(), context);
                    result.addData(salesData);
                    aggregatedCount++;
                } catch (Exception e) {
                    result.addError(0, "聚合失败[" + entry.getKey() + "]: " + e.getMessage());
                }
            }
            
            result.setMergedRows(grouped.size());
            
        } catch (IOException e) {
            result.addError(0, "读取文件失败: " + e.getMessage());
        }
        
        return result;
    }
    
    /**
     * 构建表头索引映射
     */
    private Map<Integer, String> buildHeaderIndexMap(List<String> headers) {
        Map<Integer, String> map = new HashMap<>();
        for (int i = 0; i < headers.size(); i++) {
            String header = headers.get(i).trim();
            String standardField = HEADER_MAPPING.get(header);
            if (standardField != null) {
                map.put(i, standardField);
            }
        }
        return map;
    }
    
    /**
     * 解析单行数据
     */
    private ErpSettlementRow parseRow(CSVRecord record, Map<Integer, String> headerIndexMap, int rowNum) {
        ErpSettlementRow row = new ErpSettlementRow();
        row.setRowNum(rowNum);
        
        for (Map.Entry<Integer, String> entry : headerIndexMap.entrySet()) {
            int index = entry.getKey();
            String field = entry.getValue();
            
            if (index < record.size()) {
                String value = record.get(index).trim();
                // 清理结算编号开头的单引号
                if ("settlementId".equals(field) && value.startsWith("'")) {
                    value = value.substring(1);
                }
                setFieldValue(row, field, value);
            }
        }
        
        // 校验必要字段
        if (row.getOrderId() == null || row.getOrderId().isEmpty()) {
            return null;
        }
        
        return row;
    }
    
    /**
     * 设置字段值
     */
    private void setFieldValue(ErpSettlementRow row, String field, String value) {
        switch (field) {
            case "settlementId" -> row.setSettlementId(value);
            case "orderId" -> row.setOrderId(value);
            case "storeName" -> row.setStoreName(value);
            case "siteCode" -> row.setSiteCode(value);
            case "fulfillment" -> row.setFulfillment(value);
            case "source" -> row.setSource(value);
            case "transactionType" -> row.setTransactionType(value);
            case "transactionDate" -> {
                if (value != null && !value.isEmpty()) {
                    row.setTransactionDate(parseDateTime(value));
                }
            }
            case "currencyCode" -> row.setCurrencyCode(value);
            case "amount" -> {
                if (value != null && !value.isEmpty()) {
                    row.setAmount(new BigDecimal(value));
                }
            }
            case "quantity" -> {
                if (value != null && !value.isEmpty()) {
                    row.setQuantity(Integer.parseInt(value));
                }
            }
            case "sku" -> row.setSku(value);
            case "description" -> row.setDescription(value);
            case "settlementStatus" -> row.setSettlementStatus(value);
            case "transferStatus" -> row.setTransferStatus(value);
        }
    }
    
    /**
     * 解析日期时间
     */
    private LocalDateTime parseDateTime(String value) {
        try {
            DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");
            return LocalDateTime.parse(value, formatter);
        } catch (Exception e) {
            log.warn("日期解析失败: {}", value);
            return null;
        }
    }
    
    /**
     * 聚合多行明细为一条 SalesData
     */
    private SalesData aggregateRows(List<ErpSettlementRow> rows, ParseContext context) {
        SalesData data = new SalesData();
        ErpSettlementRow first = rows.get(0);
        
        // 基础信息取第一行
        data.setImportBatchId(context.getImportBatchId());
        data.setSourceType(SalesSourceType.ERP_SETTLEMENT.getCode());
        data.setOrderId(first.getOrderId());
        data.setSiteCode(first.getSiteCode());
        data.setSku(first.getSku());
        data.setDescription(first.getDescription());
        data.setTransactionDate(first.getTransactionDate());
        data.setCurrencyCode(first.getCurrencyCode());
        data.setFulfillment(first.getFulfillment());
        data.setQuantity(first.getQuantity());
        data.setSettlementId(first.getSettlementId());
        
        // ERP特有字段
        data.setStoreName(first.getStoreName());
        data.setSettlementStatus(first.getSettlementStatus());
        data.setTransferStatus(first.getTransferStatus());
        
        // 根据来源确定交易分类
        String source = first.getSource();
        data.setTransactionType(source);
        data.setTransactionCategory(SOURCE_TO_CATEGORY.getOrDefault(source, "other"));
        
        // 初始化金额字段
        Map<String, BigDecimal> amounts = new HashMap<>();
        amounts.put("productSales", BigDecimal.ZERO);
        amounts.put("productSalesTax", BigDecimal.ZERO);
        amounts.put("shippingCredits", BigDecimal.ZERO);
        amounts.put("shippingCreditsTax", BigDecimal.ZERO);
        amounts.put("giftWrapCredits", BigDecimal.ZERO);
        amounts.put("giftWrapCreditsTax", BigDecimal.ZERO);
        amounts.put("promotionalRebates", BigDecimal.ZERO);
        amounts.put("promotionalRebatesTax", BigDecimal.ZERO);
        amounts.put("marketplaceWithheldTax", BigDecimal.ZERO);
        amounts.put("sellingFees", BigDecimal.ZERO);
        amounts.put("fbaFees", BigDecimal.ZERO);
        amounts.put("otherTransactionFees", BigDecimal.ZERO);
        amounts.put("other", BigDecimal.ZERO);
        
        // 按交易类型累加金额
        for (ErpSettlementRow row : rows) {
            String transactionType = row.getTransactionType();
            String targetField = TYPE_TO_FIELD.get(transactionType);
            
            if (targetField != null && row.getAmount() != null) {
                amounts.merge(targetField, row.getAmount(), BigDecimal::add);
            } else if (row.getAmount() != null) {
                // 未知类型归入other
                amounts.merge("other", row.getAmount(), BigDecimal::add);
            }
        }
        
        // 设置金额字段
        data.setProductSales(amounts.get("productSales"));
        data.setProductSalesTax(amounts.get("productSalesTax"));
        data.setShippingCredits(amounts.get("shippingCredits"));
        data.setShippingCreditsTax(amounts.get("shippingCreditsTax"));
        data.setGiftWrapCredits(amounts.get("giftWrapCredits"));
        data.setGiftWrapCreditsTax(amounts.get("giftWrapCreditsTax"));
        data.setPromotionalRebates(amounts.get("promotionalRebates"));
        data.setPromotionalRebatesTax(amounts.get("promotionalRebatesTax"));
        data.setMarketplaceWithheldTax(amounts.get("marketplaceWithheldTax"));
        data.setSellingFees(amounts.get("sellingFees"));
        data.setFbaFees(amounts.get("fbaFees"));
        data.setOtherTransactionFees(amounts.get("otherTransactionFees"));
        data.setOther(amounts.get("other"));
        
        // 计算合计
        data.setTotal(calculateTotal(amounts));
        
        return data;
    }
    
    /**
     * 计算合计金额
     */
    private BigDecimal calculateTotal(Map<String, BigDecimal> amounts) {
        return amounts.values().stream()
            .reduce(BigDecimal.ZERO, BigDecimal::add);
    }
    
    /**
     * 解析表头行
     */
    private List<String> parseHeaders(String headerLine) {
        if (headerLine == null) {
            return Collections.emptyList();
        }
        return Arrays.asList(headerLine.split(","));
    }
}
```

```java
package com.musheng.business.sales.service.impl;

// ... 省略导入

@Slf4j
@Service
@RequiredArgsConstructor
public class SalesDataServiceImpl implements SalesDataService {
    
    private final SalesDataMapper salesDataMapper;
    private final SalesDataParserFactory parserFactory;
    private final FieldMappingTemplateMapper templateMapper;
    private final ImportRecordMapper importRecordMapper;
    private final FileStorageService fileStorageService;
    
    @Override
    public Map<String, Object> uploadFile(MultipartFile file, String siteCode, SalesSourceType sourceType) {
        log.info("上传销售数据文件: siteCode={}, sourceType={}, fileName={}", 
                siteCode, sourceType, file.getOriginalFilename());
        
        // 1. 保存文件
        String fileId = fileStorageService.store(file, "sales");
        
        // 2. 解析表头
        SalesDataParser parser = parserFactory.getParser(sourceType);
        List<String> headers = parser.parseHeaders(file);
        
        // 3. 读取样本数据（前3行）
        // ... 省略
        
        Map<String, Object> result = new HashMap<>();
        result.put("fileId", fileId);
        result.put("fileName", file.getOriginalFilename());
        result.put("fileSize", file.getSize());
        result.put("headers", headers);
        
        return result;
    }
    
    @Override
    public ParseResult previewImport(SalesImportRequest request) {
        log.info("预览销售数据导入: {}", request);
        
        // 1. 获取文件
        InputStream inputStream = fileStorageService.load(request.getFileId());
        
        // 2. 获取映射模板
        FieldMappingTemplate template = templateMapper.selectById(request.getTemplateId());
        Map<String, String> fieldMapping = parseFieldMapping(template.getMappingConfig());
        
        // 3. 构建解析上下文
        ParseContext context = ParseContext.builder()
            .inputStream(inputStream)
            .siteCode(request.getSiteCode())
            .sourceType(request.getSourceType())
            .template(template)
            .fieldMapping(fieldMapping)
            .build();
        
        // 4. 预览解析
        SalesDataParser parser = parserFactory.getParser(request.getSourceType());
        return parser.preview(context, 10);
    }
    
    @Override
    @Transactional(rollbackFor = Exception.class)
    public Map<String, Object> executeImport(SalesImportRequest request) {
        log.info("执行销售数据导入: {}", request);
        
        // 1. 创建导入记录
        ImportRecord importRecord = new ImportRecord();
        importRecord.setBatchNo(generateBatchNo());
        importRecord.setDataType("sales");
        importRecord.setFileName(request.getFileName());
        importRecord.setImportStatus("processing");
        importRecordMapper.insert(importRecord);
        
        try {
            // 2. 获取文件和模板
            InputStream inputStream = fileStorageService.load(request.getFileId());
            FieldMappingTemplate template = templateMapper.selectById(request.getTemplateId());
            Map<String, String> fieldMapping = parseFieldMapping(template.getMappingConfig());
            
            // 3. 构建解析上下文
            ParseContext context = ParseContext.builder()
                .inputStream(inputStream)
                .siteCode(request.getSiteCode())
                .sourceType(request.getSourceType())
                .template(template)
                .fieldMapping(fieldMapping)
                .importBatchId(importRecord.getId())
                .build();
            
            // 4. 完整解析
            SalesDataParser parser = parserFactory.getParser(request.getSourceType());
            ParseResult parseResult = parser.parse(context);
            
            // 5. 去重校验 + 批量入库
            int successCount = 0;
            int failCount = parseResult.getFailCount();
            List<String> duplicates = new ArrayList<>();
            
            for (SalesData data : parseResult.getDataList()) {
                if (isDuplicate(data)) {
                    failCount++;
                    duplicates.add(data.getOrderId());
                } else {
                    salesDataMapper.insert(data);
                    successCount++;
                }
            }
            
            // 6. 更新导入记录
            importRecord.setTotalCount(parseResult.getTotalRows());
            importRecord.setSuccessCount(successCount);
            importRecord.setFailCount(failCount);
            importRecord.setImportStatus(failCount == 0 ? "success" : 
                    (successCount > 0 ? "partial" : "fail"));
            importRecord.setCompleteTime(LocalDateTime.now());
            
            if (!duplicates.isEmpty()) {
                importRecord.setErrorMessage("重复订单: " + 
                    String.join(", ", duplicates.subList(0, Math.min(10, duplicates.size()))));
            }
            
            importRecordMapper.updateById(importRecord);
            
            // 7. 返回结果
            Map<String, Object> result = new HashMap<>();
            result.put("batchNo", importRecord.getBatchNo());
            result.put("totalCount", parseResult.getTotalRows());
            result.put("mergedCount", parseResult.getMergedRows());
            result.put("successCount", successCount);
            result.put("failCount", failCount);
            
            return result;
            
        } catch (Exception e) {
            log.error("导入失败", e);
            importRecord.setImportStatus("fail");
            importRecord.setErrorMessage(e.getMessage());
            importRecordMapper.updateById(importRecord);
            throw new BusinessException(ErrorCode.IMPORT_FAILED, e.getMessage());
        }
    }
    
    /**
     * 检查是否重复
     */
    private boolean isDuplicate(SalesData data) {
        LambdaQueryWrapper<SalesData> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(SalesData::getOrderId, data.getOrderId())
               .eq(SalesData::getSiteCode, data.getSiteCode())
               .eq(SalesData::getSku, data.getSku())
               .eq(SalesData::getTransactionCategory, data.getTransactionCategory());
        return salesDataMapper.selectCount(wrapper) > 0;
    }
    
    /**
     * 解析字段映射配置
     */
    private Map<String, String> parseFieldMapping(String mappingConfig) {
        if (mappingConfig == null || mappingConfig.isEmpty()) {
            return Collections.emptyMap();
        }
        try {
            return objectMapper.readValue(mappingConfig, 
                    new TypeReference<Map<String, String>>() {});
        } catch (Exception e) {
            log.warn("解析映射配置失败", e);
            return Collections.emptyMap();
        }
    }
    
    /**
     * 生成批次号
     */
    private String generateBatchNo() {
        return "SALES-" + System.currentTimeMillis() + "-" + 
               UUID.randomUUID().toString().substring(0, 8);
    }
}
```

```sql
    ADD COLUMN transfer_status VARCHAR(32) COMMENT '转账状态（ERP数据）';

-- 字段映射模板表增加字段
ALTER TABLE t_field_mapping_template
    ADD COLUMN source_type VARCHAR(32) COMMENT '数据源类型: AMAZON_ORIGINAL/ERP_SETTLEMENT';

-- 更新索引
CREATE INDEX idx_source_type ON t_sales_data(source_type);
```
