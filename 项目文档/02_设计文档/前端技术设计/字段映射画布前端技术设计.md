## 1. 设计概述

### 1.1 背景

基于《字段映射画布交互设计文档》，本文档详细描述字段映射画布功能的前端技术实现方案，包括组件结构、状态管理、交互实现、API对接等内容。

### 1.2 目标

- 实现可视化字段映射画布，支持拖拽/点击建立映射
- 支持文件上传解析和粘贴表头两种源字段获取方式
- 支持数据类型联动加载目标字段
- 支持智能映射、默认值配置等辅助功能
- 良好的用户体验和性能表现

### 1.3 技术栈

| 类别 | 技术选型 | 版本 | 用途 |
|------|---------|------|------|
| 框架 | Vue.js | 3.4+ | 组件开发 |
| 语言 | TypeScript | 5.x | 类型安全 |
| UI组件 | Ant Design Vue | 4.x | 基础UI组件 |
| 拖拽 | VueDraggable Plus | 1.x | 字段拖拽 |
| 文件解析 | xlsx + PapaParse | 0.18+ / 5.x | Excel/CSV解析 |
| SVG | 原生SVG | - | 连线绘制 |

## 3. 类型定义

### 3.1 核心类型

```typescript
// types.ts

/** 数据类型枚举 */
export type FieldMappingDataType = 'SALES' | 'SHIPPING' | 'ADVERTISING' | 'RATE'

/** 销售数据子类型 */
export type SalesSubType = 'ORIGINAL' | 'ERP'

/** 源字段 */
export interface SourceField {
  /** 字段名 */
  name: string
  /** 示例值 */
  sample?: string
  /** 字段索引（在文件中的列索引） */
  index: number
  /** 是否已映射 */
  mapped?: boolean
}

/** 目标字段 */
export interface TargetField {
  /** 字段名（数据库字段） */
  field: string
  /** 中文标签 */
  label: string
  /** 字段描述 */
  description?: string
  /** 数据类型 */
  type: 'string' | 'number' | 'datetime' | 'boolean'
  /** 是否必填 */
  required: boolean
  /** 最大长度（string类型） */
  maxLength?: number
  /** 精度（number类型） */
  precision?: number
  /** 默认值 */
  defaultValue?: string | number | boolean
  /** 映射的源字段名 */
  mappedSource?: string
}

/** 映射配置 */
export interface MappingConfig {
  /** 源字段名 */
  source: string
  /** 目标字段名 */
  target: string
  /** 源字段索引 */
  sourceIndex?: number
}

/** 默认值配置 */
export interface DefaultValueConfig {
  /** 目标字段名 */
  field: string
  /** 默认值 */
  value: string | number | boolean
}

/** 完整模板配置 */
export interface FieldMappingTemplate {
  /** 模板ID */
  id?: number
  /** 模板名称 */
  name: string
  /** 站点编码 */
  siteCode: string
  /** 数据类型 */
  dataType: FieldMappingDataType
  /** 子类型（销售数据） */
  subType?: SalesSubType
  /** 字段映射列表 */
  mappings: MappingConfig[]
  /** 默认值列表 */
  defaultValues: DefaultValueConfig[]
  /** 表头行号（保存用于下次编辑） */
  headerRow?: number
  /** 源字段列表（可选保存） */
  sourceFields?: SourceField[]
}
```

## 4. 组件实现

### 4.1 画布主组件 (index.vue)

```vue
<template>
  <div class="field-mapping-canvas">
    <!-- 工具栏 -->
    <MappingToolbar
      :mapped-count="mappedCount"
      :total-count="targetFields.length"
      :has-source-fields="sourceFields.length > 0"
      @auto-match="handleAutoMatch"
      @reset="handleReset"
      @clear-source="handleClearSource"
    />

    <!-- 画布区域 -->
    <div class="canvas-body" ref="canvasRef">
      <!-- 左侧：源字段列表 -->
      <div class="source-panel">
        <div class="panel-header">
          <span class="panel-title">源字段 (CSV)</span>
          <span class="field-count">{{ sourceFields.length }} 个</span>
        </div>
        <SourceFieldList
          :fields="sourceFields"
          :selected-field="selectedSourceField"
          :mappings="mappings"
          :allow-delete="allowSourceDelete"
          @select="handleSourceSelect"
          @delete="handleSourceDelete"
          @dragstart="handleDragStart"
          @dragend="handleDragEnd"
        />
      </div>

      <!-- 中间：连线区域 -->
      <div class="mapping-area" ref="mappingAreaRef">
        <MappingLines
          :mappings="mappings"
          :source-positions="sourcePositions"
          :target-positions="targetPositions"
          :selected-mapping="selectedMapping"
          :dragging-line="draggingLine"
          @select="handleMappingSelect"
          @delete="handleMappingDelete"
        />
      </div>

      <!-- 右侧：目标字段列表 -->
      <div class="target-panel">
        <div class="panel-header">
          <span class="panel-title">目标字段 (数据库)</span>
          <span class="field-count">
            已映射 {{ mappedCount }}/{{ targetFields.length }}
          </span>
        </div>
        <TargetFieldList
          :fields="targetFields"
          :selected-field="selectedTargetField"
          :mappings="mappings"
          :default-values="defaultValues"
          @select="handleTargetSelect"
          @set-default="handleSetDefault"
          @drop="handleDrop"
          @dragover="handleDragOver"
        />
      </div>
    </div>

    <!-- 必填未映射警告 -->
    <div v-if="unmappedRequiredFields.length > 0" class="mapping-warning">
      <WarningOutlined />
      <span>
        还有 {{ unmappedRequiredFields.length }} 个必填字段未映射：
        {{ unmappedRequiredFields.map(f => f.label).join('、') }}
      </span>
    </div>

    <!-- 默认值设置弹窗 -->
    <DefaultValueModal
      v-model:visible="defaultValueModalVisible"
      :field="currentDefaultValueField"
      @confirm="handleDefaultValueConfirm"
    />

    <!-- 智能匹配预览弹窗 -->
    <AutoMatchPreview
      v-model:visible="autoMatchPreviewVisible"
      :result="autoMatchResult"
      @apply="handleApplyAutoMatch"
    />
  </div>
</template>

<script setup lang="ts">
import { ref, computed, watch, provide } from 'vue'
import { WarningOutlined } from '@ant-design/icons-vue'
import MappingToolbar from './MappingToolbar.vue'
import SourceFieldList from './SourceFieldList.vue'
import TargetFieldList from './TargetFieldList.vue'
import MappingLines from './MappingLines.vue'
import DefaultValueModal from './DefaultValueModal.vue'
import AutoMatchPreview from './AutoMatchPreview.vue'
import { useDragMapping } from './hooks/useDragMapping'
import { useClickMapping } from './hooks/useClickMapping'
import { useLineCalculation } from './hooks/useLineCalculation'
import { useAutoMatch } from './hooks/useAutoMatch'
import type {
  SourceField,
  TargetField,
  MappingConfig,
  DefaultValueConfig,
  FieldMappingDataType,
  SalesSubType,
  AutoMatchResult
} from './types'

// Props
const props = withDefaults(defineProps<{
  sourceFields?: SourceField[]
  targetFields?: TargetField[]
  initialMappings?: MappingConfig[]
  initialDefaultValues?: DefaultValueConfig[]
  dataType?: FieldMappingDataType
  subType?: SalesSubType
  siteCode?: string
  readonly?: boolean
  allowSourceDelete?: boolean
}>(), {
  sourceFields: () => [],
  targetFields: () => [],
  initialMappings: () => [],
  initialDefaultValues: () => [],
  readonly: false,
  allowSourceDelete: true
})

// Emits
const emit = defineEmits<{
  (e: 'update:mappings', mappings: MappingConfig[]): void
  (e: 'update:defaultValues', values: DefaultValueConfig[]): void
  (e: 'sourceDelete', field: SourceField): void
}>()

// 响应式状态
const mappings = ref<MappingConfig[]>([...props.initialMappings])
const defaultValues = ref<DefaultValueConfig[]>([...props.initialDefaultValues])

// 拖拽相关
const {
  draggingLine,
  handleDragStart,
  handleDragEnd,
  handleDrop,
  handleDragOver
} = useDragMapping(mappings, props.sourceFields, props.targetFields)

// 点击选择相关
const {
  selectedSourceField,
  selectedTargetField,
  selectedMapping,
  handleSourceSelect,
  handleTargetSelect,
  handleMappingSelect,
  clearSelection
} = useClickMapping(mappings, emit)

// 连线计算
const canvasRef = ref<HTMLElement>()
const mappingAreaRef = ref<HTMLElement>()
const { sourcePositions, targetPositions, recalculatePositions } = useLineCalculation(
  canvasRef,
  mappingAreaRef,
  props.sourceFields,
  props.targetFields
)

// 智能匹配
const autoMatchPreviewVisible = ref(false)
const autoMatchResult = ref<AutoMatchResult | null>(null)
const { executeAutoMatch } = useAutoMatch()

// 默认值弹窗
const defaultValueModalVisible = ref(false)
const currentDefaultValueField = ref<TargetField | null>(null)

// 计算属性
const mappedCount = computed(() => {
  return mappings.value.length + defaultValues.value.length
})

const unmappedRequiredFields = computed(() => {
  return props.targetFields.filter(field => {
    if (!field.required) return false
    const hasMMapping = mappings.value.some(m => m.target === field.field)
    const hasDefault = defaultValues.value.some(d => d.field === field.field)
    return !hasMMapping && !hasDefault
  })
})

// 事件处理
const handleMappingDelete = (mapping: MappingConfig) => {
  mappings.value = mappings.value.filter(
    m => !(m.source === mapping.source && m.target === mapping.target)
  )
  emit('update:mappings', mappings.value)
}

const handleSourceDelete = (field: SourceField) => {
  // 删除相关映射
  mappings.value = mappings.value.filter(m => m.source !== field.name)
  emit('update:mappings', mappings.value)
  emit('sourceDelete', field)
}

const handleSetDefault = (field: TargetField) => {
  currentDefaultValueField.value = field
  defaultValueModalVisible.value = true
}

const handleDefaultValueConfirm = (value: string | number | boolean) => {
  if (!currentDefaultValueField.value) return
  
  const existing = defaultValues.value.findIndex(
    d => d.field === currentDefaultValueField.value!.field
  )
  
  if (existing >= 0) {
    defaultValues.value[existing].value = value
  } else {
    defaultValues.value.push({
      field: currentDefaultValueField.value.field,
      value
    })
  }
  
  emit('update:defaultValues', defaultValues.value)
  defaultValueModalVisible.value = false
}

const handleAutoMatch = async () => {
  const result = await executeAutoMatch(
    props.sourceFields,
    props.targetFields,
    props.siteCode
  )
  autoMatchResult.value = result
  autoMatchPreviewVisible.value = true
}

const handleApplyAutoMatch = (selectedMappings: MappingConfig[]) => {
  mappings.value = [...mappings.value, ...selectedMappings]
  emit('update:mappings', mappings.value)
  autoMatchPreviewVisible.value = false
}

const handleReset = () => {
  mappings.value = []
  defaultValues.value = []
  clearSelection()
  emit('update:mappings', [])
  emit('update:defaultValues', [])
}

const handleClearSource = () => {
  mappings.value = []
  emit('update:mappings', [])
  // 通知父组件清空源字段
}

// 监听props变化
watch(() => props.initialMappings, (val) => {
  mappings.value = [...val]
}, { deep: true })

watch(() => props.initialDefaultValues, (val) => {
  defaultValues.value = [...val]
}, { deep: true })

// 字段变化时重新计算位置
watch([() => props.sourceFields, () => props.targetFields], () => {
  recalculatePositions()
}, { deep: true })

// 提供给子组件的上下文
provide('canvasReadonly', props.readonly)
</script>
```

```vue
import { InboxOutlined, FileExcelOutlined } from '@ant-design/icons-vue'

import { useFieldParser } from './hooks/useFieldParser'

const emit = defineEmits<{
  (e: 'parsed', fields: SourceField[]): void

const uploadedFile = ref<File | null>(null)

  // 大小校验
  if (file.size > 10 * 1024 * 1024) {
    message.error('文件大小不能超过 10MB')
    return false
  }
  
  uploadedFile.value = file
  // 自动打开预览
  handlePreviewFile()
  return false
}

// 预览文件
const handlePreviewFile = async () => {
  if (!uploadedFile.value) return
  
  previewLoading.value = true
  previewModalVisible.value = true
  
  try {
    const result = await previewFile(uploadedFile.value, 10)
    filePreviewData.value = result
  } catch (error) {
    message.error('文件预览失败')
  } finally {
    previewLoading.value = false
  }
}

// 确认预览选择
const handlePreviewConfirm = async (headerRow: number, sheetName?: string) => {
  if (!uploadedFile.value) return
  
  try {
    const result = await parseFile(uploadedFile.value, headerRow, sheetName)
    emit('parsed', result.fields)
    previewModalVisible.value = false
    message.success(`成功解析 ${result.fields.length} 个字段`)
  } catch (error) {
    message.error('字段解析失败')
  }
}

// 粘贴解析
const handleParsePaste = () => {
  if (!pasteContent.value.trim()) {
    message.warning('请先粘贴表头内容')
    return
  }
  
  try {
    const fields = parseHeaderText(pasteContent.value, delimiter.value)
    
    if (fields.length === 0) {
      message.warning('未检测到有效的表头字段')
      return
    }
    
    if (fields.length === 1) {
      message.warning('检测到字段过少，请确认分隔符是否正确')
    }
    
    emit('parsed', fields)
    message.success(`成功解析 ${fields.length} 个字段`)
  } catch (error) {
    message.error('解析失败')
  }
}
```

### 4.3 文件解析 Hook (useFieldParser.ts)

```typescript
// hooks/useFieldParser.ts
import * as XLSX from 'xlsx'
import Papa from 'papaparse'
import type { 
  SourceField, 
  FilePreviewResult, 
  FilePreviewRow,
  ParsedFieldsResult,
  DelimiterType 
} from '../types'

export function useFieldParser() {
  /**
   * 预览文件（获取前N行）
   */
  const previewFile = async (
    file: File, 
    previewRows: number = 10
  ): Promise<FilePreviewResult> => {
    const extension = file.name.split('.').pop()?.toLowerCase()
    
    if (extension === 'csv') {
      return previewCsvFile(file, previewRows)
    } else if (['xlsx', 'xls'].includes(extension || '')) {
      return previewExcelFile(file, previewRows)
    } else {
      throw new Error('不支持的文件格式')
    }
  }

  /**
   * 预览 CSV 文件
   */
  const previewCsvFile = (
    file: File, 
    previewRows: number
  ): Promise<FilePreviewResult> => {
    return new Promise((resolve, reject) => {
      let rowCount = 0
      const rows: FilePreviewRow[] = []
      let detectedDelimiter = ','

      Papa.parse(file, {
        preview: previewRows,
        step: (results, parser) => {
          rowCount++
          const cells = results.data as string[]
          rows.push({
            rowNum: rowCount,
            content: cells.join('\t'),
            cells
          })
          
          // 检测分隔符
          if (rowCount === 1) {
            detectedDelimiter = results.meta.delimiter || ','
          }
        },
        complete: () => {
          resolve({
            rows,
            totalRows: rowCount,
            encoding: 'UTF-8',
            delimiter: detectedDelimiter
          })
        },
        error: (error) => {
          reject(error)
        }
      })
    })
  }

  /**
   * 预览 Excel 文件
   */
  const previewExcelFile = async (
    file: File, 
    previewRows: number
  ): Promise<FilePreviewResult> => {
    const arrayBuffer = await file.arrayBuffer()
    const workbook = XLSX.read(arrayBuffer, { type: 'array' })
    
    const sheets = workbook.SheetNames
    const sheetName = sheets[0]
    const worksheet = workbook.Sheets[sheetName]
    
    // 转换为 JSON，保留前 N 行
    const jsonData = XLSX.utils.sheet_to_json<string[]>(worksheet, {
      header: 1,
      defval: '',
      range: 0
    }).slice(0, previewRows)

    const rows: FilePreviewRow[] = jsonData.map((row, index) => ({
      rowNum: index + 1,
      content: row.join('\t'),
      cells: row.map(String)
    }))

    // 获取总行数
    const range = XLSX.utils.decode_range(worksheet['!ref'] || 'A1')
    const totalRows = range.e.r + 1

    return {
      rows,
      totalRows,
      encoding: 'UTF-8',
      delimiter: '\t',
      sheetName,
      sheets
    }
  }

  /**
   * 解析文件获取字段
   */
  const parseFile = async (
    file: File,
    headerRow: number = 1,
    sheetName?: string
  ): Promise<ParsedFieldsResult> => {
    const extension = file.name.split('.').pop()?.toLowerCase()
    
    if (extension === 'csv') {
      return parseCsvFields(file, headerRow)
    } else if (['xlsx', 'xls'].includes(extension || '')) {
      return parseExcelFields(file, headerRow, sheetName)
    } else {
      throw new Error('不支持的文件格式')
    }
  }

  /**
   * 解析 CSV 字段
   */
  const parseCsvFields = (
    file: File, 
    headerRow: number
  ): Promise<ParsedFieldsResult> => {
    return new Promise((resolve, reject) => {
      let currentRow = 0
      let headerFields: string[] = []
      let sampleValues: string[] = []
      let detectedDelimiter = ','

      Papa.parse(file, {
        step: (results, parser) => {
          currentRow++
          
          if (currentRow === headerRow) {
            headerFields = results.data as string[]
            detectedDelimiter = results.meta.delimiter || ','
          } else if (currentRow === headerRow + 1) {
            sampleValues = results.data as string[]
            parser.abort()
          }
        },
        complete: () => {
          const fields: SourceField[] = headerFields
            .filter(name => name.trim() !== '')
            .map((name, index) => ({
              name: name.trim(),
              sample: sampleValues[index]?.substring(0, 30) || '',
              index
            }))

          resolve({
            fields,
            totalColumns: headerFields.length,
            headerRow,
            encoding: 'UTF-8',
            delimiter: detectedDelimiter
          })
        },
        error: reject
      })
    })
  }

  /**
   * 解析 Excel 字段
   */
  const parseExcelFields = async (
    file: File,
    headerRow: number,
    sheetName?: string
  ): Promise<ParsedFieldsResult> => {
    const arrayBuffer = await file.arrayBuffer()
    const workbook = XLSX.read(arrayBuffer, { type: 'array' })
    
    const sheet = sheetName || workbook.SheetNames[0]
    const worksheet = workbook.Sheets[sheet]
    
    const jsonData = XLSX.utils.sheet_to_json<string[]>(worksheet, {
      header: 1,
      defval: '',
      range: headerRow - 1
    })

    const headerFields = jsonData[0] || []
    const sampleValues = jsonData[1] || []

    const fields: SourceField[] = headerFields
      .filter((name: string) => name?.toString().trim() !== '')
      .map((name: string, index: number) => ({
        name: name.toString().trim(),
        sample: sampleValues[index]?.toString().substring(0, 30) || '',
        index
      }))

    return {
      fields,
      totalColumns: headerFields.length,
      headerRow,
      encoding: 'UTF-8',
      delimiter: '\t'
    }
  }

  /**
   * 解析粘贴的表头文本
   */
  const parseHeaderText = (
    text: string, 
    delimiterType: DelimiterType
  ): SourceField[] => {
    let delimiter: string | RegExp

    if (delimiterType === 'auto') {
      delimiter = detectDelimiter(text)
    } else {
      const delimiterMap: Record<DelimiterType, string | RegExp> = {
        auto: '\t',
        tab: '\t',
        comma: ',',
        semicolon: ';',
        pipe: '|',
        space: /\s+/
      }
      delimiter = delimiterMap[delimiterType]
    }

    const fields = text.split(delimiter)
      .map(s => s.trim())
      .filter(s => s !== '')

    return fields.map((name, index) => ({
      name,
      index,
      sample: undefined
    }))
  }

  /**
   * 自动检测分隔符
   */
  const detectDelimiter = (text: string): string => {
    if (text.includes('\t')) return '\t'
    
    const commaCount = (text.match(/,/g) || []).length
    const semicolonCount = (text.match(/;/g) || []).length
    const pipeCount = (text.match(/\|/g) || []).length

    if (commaCount > 2) return ','
    if (semicolonCount > 2) return ';'
    if (pipeCount > 2) return '|'
    
    return ' '
  }

  return {
    previewFile,
    parseFile,
    parseHeaderText,
    detectDelimiter
  }
}
```

### 4.4 智能匹配 Hook (useAutoMatch.ts)

```typescript
// hooks/useAutoMatch.ts
import { autoMatchFields } from '@/api/fieldMapping'
import type { 
  SourceField, 
  TargetField, 
  MappingConfig,
  AutoMatchResult,
  MatchType 
} from '../types'

export function useAutoMatch() {
  /**
   * 执行智能匹配
   * 优先使用前端匹配，可选调用后端API
   */
  const executeAutoMatch = async (
    sourceFields: SourceField[],
    targetFields: TargetField[],
    siteCode?: string,
    useBackend: boolean = false
  ): Promise<AutoMatchResult> => {
    if (useBackend) {
      // 调用后端API
      const response = await autoMatchFields({
        dataType: 'SALES',
        sourceFields: sourceFields.map(f => f.name),
        siteCode
      })
      return response.data
    }

    // 前端匹配逻辑
    return localAutoMatch(sourceFields, targetFields)
  }

  /**
   * 前端本地匹配算法
   */
  const localAutoMatch = (
    sourceFields: SourceField[],
    targetFields: TargetField[]
  ): AutoMatchResult => {
    const mappings: Array<{
      source: string
      target: string
      confidence: number
      matchType: MatchType
    }> = []

    const matchedSources = new Set<string>()
    const matchedTargets = new Set<string>()

    // 按优先级依次尝试匹配
    const matchers: Array<{
      type: MatchType
      match: (source: string, target: TargetField) => boolean
      confidence: number
    }> = [
      // 1. 完全匹配
      {
        type: 'exact',
        match: (s, t) => s === t.field,
        confidence: 1.0
      },
      // 2. 忽略大小写
      {
        type: 'ignore_case',
        match: (s, t) => s.toLowerCase() === t.field.toLowerCase(),
        confidence: 0.95
      },
      // 3. 忽略分隔符
      {
        type: 'ignore_separator',
        match: (s, t) => normalize(s) === normalize(t.field),
        confidence: 0.9
      },
      // 4. 中文标签匹配
      {
        type: 'label',
        match: (s, t) => s === t.label || normalize(s) === normalize(t.label),
        confidence: 0.85
      },
      // 5. 包含匹配
      {
        type: 'contains',
        match: (s, t) => {
          const ns = normalize(s)
          const nt = normalize(t.field)
          return ns.includes(nt) || nt.includes(ns)
        },
        confidence: 0.75
      },
      // 6. 相似度匹配
      {
        type: 'similar',
        match: (s, t) => calculateSimilarity(s, t.field) > 0.7,
        confidence: 0.6
      }
    ]

    // 遍历匹配器
    for (const matcher of matchers) {
      for (const source of sourceFields) {
        if (matchedSources.has(source.name)) continue

        for (const target of targetFields) {
          if (matchedTargets.has(target.field)) continue

          if (matcher.match(source.name, target)) {
            mappings.push({
              source: source.name,
              target: target.field,
              confidence: matcher.confidence,
              matchType: matcher.type
            })
            matchedSources.add(source.name)
            matchedTargets.add(target.field)
            break
          }
        }
      }
    }

    // 未匹配的字段
    const unmatchedSource = sourceFields
      .filter(f => !matchedSources.has(f.name))
      .map(f => f.name)

    const unmatchedTarget = targetFields
      .filter(f => !matchedTargets.has(f.field))
      .map(f => f.field)

    return {
      mappings,
      unmatchedSource,
      unmatchedTarget
    }
  }

  /**
   * 规范化字段名（移除分隔符）
   */
  const normalize = (str: string): string => {
    return str.toLowerCase().replace(/[-_\s.]/g, '')
  }

  /**
   * 计算字符串相似度（Levenshtein 距离）
   */
  const calculateSimilarity = (s1: string, s2: string): number => {
    const a = s1.toLowerCase()
    const b = s2.toLowerCase()
    
    if (a === b) return 1
    if (a.length === 0 || b.length === 0) return 0

    const matrix: number[][] = []

    for (let i = 0; i <= b.length; i++) {
      matrix[i] = [i]
    }
    for (let j = 0; j <= a.length; j++) {
      matrix[0][j] = j
    }

    for (let i = 1; i <= b.length; i++) {
      for (let j = 1; j <= a.length; j++) {
        if (b.charAt(i - 1) === a.charAt(j - 1)) {
          matrix[i][j] = matrix[i - 1][j - 1]
        } else {
          matrix[i][j] = Math.min(
            matrix[i - 1][j - 1] + 1,
            matrix[i][j - 1] + 1,
            matrix[i - 1][j] + 1
          )
        }
      }
    }

    const distance = matrix[b.length][a.length]
    const maxLength = Math.max(a.length, b.length)
    return 1 - distance / maxLength
  }

  return {
    executeAutoMatch,
    localAutoMatch
  }
}
```

```typescript
// stores/fieldMapping.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useFieldMappingStore = defineStore('fieldMapping', () => {
  // 当前编辑的模板
  const currentTemplate = ref<FieldMappingTemplate | null>(null)
  
  // 源字段列表
  const sourceFields = ref<SourceField[]>([])
  
  // 目标字段列表
  const targetFields = ref<TargetField[]>([])
  
  // 目标字段加载状态
  const targetFieldsLoading = ref(false)
  
  // 映射配置
  const mappings = ref<MappingConfig[]>([])
  
  // 默认值配置
  const defaultValues = ref<DefaultValueConfig[]>([])

  // 计算属性
  const canSave = computed(() => {
    if (!currentTemplate.value?.name) return false
    if (!currentTemplate.value?.siteCode) return false
    if (!currentTemplate.value?.dataType) return false
    
    // 检查必填字段是否都有映射或默认值
    const requiredFields = targetFields.value.filter(f => f.required)
    return requiredFields.every(field => {
      const hasMapping = mappings.value.some(m => m.target === field.field)
      const hasDefault = defaultValues.value.some(d => d.field === field.field)
      return hasMapping || hasDefault
    })
  })

  // 加载目标字段
  const loadTargetFields = async (
    dataType: FieldMappingDataType,
    subType?: SalesSubType
  ) => {
    targetFieldsLoading.value = true
    try {
      const response = await getTargetFields(dataType, subType)
      targetFields.value = response.data.fields
    } catch (error) {
      console.error('加载目标字段失败:', error)
      throw error
    } finally {
      targetFieldsLoading.value = false
    }
  }

  // 设置源字段
  const setSourceFields = (fields: SourceField[]) => {
    sourceFields.value = fields
    // 清空现有映射（可选）
    // mappings.value = []
  }

  // 删除源字段
  const removeSourceField = (fieldName: string) => {
    sourceFields.value = sourceFields.value.filter(f => f.name !== fieldName)
    // 同时删除相关映射
    mappings.value = mappings.value.filter(m => m.source !== fieldName)
  }

  // 更新映射
  const updateMappings = (newMappings: MappingConfig[]) => {
    mappings.value = newMappings
  }

  // 更新默认值
  const updateDefaultValues = (values: DefaultValueConfig[]) => {
    defaultValues.value = values
  }

  // 加载模板详情
  const loadTemplate = async (templateId: number) => {
    const response = await getTemplateDetail(templateId)
    const template = response.data
    
    currentTemplate.value = template
    mappings.value = template.mappings || []
    defaultValues.value = template.defaultValues || []
    
    // 加载目标字段
    await loadTargetFields(template.dataType, template.subType)
    
    // 如果有保存的源字段，恢复
    if (template.sourceFields) {
      sourceFields.value = template.sourceFields
    }
  }

  // 保存模板
  const save = async () => {
    if (!currentTemplate.value) {
      throw new Error('没有可保存的模板')
    }

    const payload: FieldMappingTemplate = {
      ...currentTemplate.value,
      mappings: mappings.value,
      defaultValues: defaultValues.value,
      sourceFields: sourceFields.value
    }

    const response = await saveTemplate(payload)
    currentTemplate.value.id = response.data.id
    return response.data
  }

  // 重置状态
  const reset = () => {
    currentTemplate.value = null
    sourceFields.value = []
    targetFields.value = []
    mappings.value = []
    defaultValues.value = []
  }

  return {
    currentTemplate,
    sourceFields,
    targetFields,
    targetFieldsLoading,
    mappings,
    defaultValues,
    canSave,
    loadTargetFields,
    setSourceFields,
    removeSourceField,
    updateMappings,
    updateDefaultValues,
    loadTemplate,
    save,
    reset
  }
})
```

## 8. 性能优化

### 8.1 虚拟滚动

当字段数量较多时（超过50个），启用虚拟滚动：

```typescript
// hooks/useVirtualScroll.ts
import { ref, computed, watch } from 'vue'

export function useVirtualScroll<T>(
  items: Ref<T[]>,
  itemHeight: number = 60,
  containerHeight: number = 400
) {
  const scrollTop = ref(0)
  const visibleCount = Math.ceil(containerHeight / itemHeight) + 2

  const visibleItems = computed(() => {
    const startIndex = Math.floor(scrollTop.value / itemHeight)
    const endIndex = startIndex + visibleCount
    
    return items.value.slice(startIndex, endIndex).map((item, index) => ({
      item,
      style: {
        position: 'absolute',
        top: `${(startIndex + index) * itemHeight}px`,
        width: '100%'
      }
    }))
  })

  const totalHeight = computed(() => items.value.length * itemHeight)

  const handleScroll = (e: Event) => {
    scrollTop.value = (e.target as HTMLElement).scrollTop
  }

  return {
    visibleItems,
    totalHeight,
    handleScroll
  }
}
```
